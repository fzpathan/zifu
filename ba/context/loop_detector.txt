from collections import Counter, deque
from typing import Any


class LoopDetector:
    def __init__(
        self,
        max_exact_repeats: int = 4,
        max_cycle_length: int = 5,
        window_size: int = 100,
        max_tool_freq: int = 5,
    ):
        self.max_exact_repeats = max_exact_repeats
        self.max_cycle_length = max_cycle_length
        self.window_size = window_size
        self.max_tool_freq = max_tool_freq
        self.history: deque = deque(maxlen=window_size)

    def _make_signature(self, action_type: str, **details: Any) -> str:
        if action_type == "tool_call":
            parts = ["tool_call", details.get("tool_name", "")]
            args = details.get("args", {})
            if isinstance(args, dict):
                for k in sorted(args.keys()):
                    parts.append(f"{k}={str(args[k])}")
            return "|".join(parts)

        # Normalize all response entries to the same token so they don't
        # pollute the window with never-matching unique text, but still
        # participate correctly in cycle detection.
        return "response"

    def record_action(self, action_type: str, **details: Any) -> None:
        self.history.append(self._make_signature(action_type, **details))

    def check_for_loop(self) -> str | None:
        if len(self.history) < 2:
            return None

        history = list(self.history)

        # 1. Consecutive exact repeats of the same action
        if len(history) >= self.max_exact_repeats:
            recent = history[-self.max_exact_repeats:]
            if len(set(recent)) == 1 and recent[0] != "response":
                return f"Same action repeated {self.max_exact_repeats} times consecutively"

        # 2. Repeating cycle of identical consecutive subsequences
        if len(history) >= self.max_cycle_length * 2:
            for cycle_len in range(2, min(self.max_cycle_length + 1, len(history) // 2 + 1)):
                recent = history[-cycle_len * 2:]
                if recent[:cycle_len] == recent[cycle_len:]:
                    # Ignore trivial all-response cycles
                    if set(recent[:cycle_len]) != {"response"}:
                        return f"Detected repeating cycle of length {cycle_len}"

        # 3. Same tool+args appearing too many times anywhere in the window
        #    (catches non-consecutive loops, e.g. reading the same file 5 times
        #    across 40 turns with other actions in between)
        tool_counts = Counter(s for s in history if s.startswith("tool_call|"))
        for sig, count in tool_counts.items():
            if count >= self.max_tool_freq:
                tool_name = sig.split("|")[1] if "|" in sig else sig
                return (
                    f"Tool '{tool_name}' called {count} times with identical "
                    f"arguments in the last {len(history)} actions"
                )

        return None

    def clear(self) -> None:
        self.history.clear()
