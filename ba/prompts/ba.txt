def get_ba_system_prompt(
    config: Config,
    user_memory: str | None = None,
    tools: list[Tool] | None = None,
) -> str:
    parts = [
        _get_identity_section(),
        _get_environment_section(config),
        _get_etl_domain_section(),
    ]

    if tools:
        parts.append(_get_tool_guidelines_section(tools))

    parts.extend(
        [
            _get_agents_md_section(),
            _get_security_section(),
        ]
    )

    if config.developer_instructions:
        parts.append(_get_developer_instructions_section(config.developer_instructions))

    if config.user_instructions:
        parts.append(_get_user_instructions_section(config.user_instructions))

    if user_memory:
        parts.append(_get_memory_section(user_memory))

    parts.append(_get_operational_section())
    parts.append(_get_ba_workflow_section())

    return "\n\n".join(parts)


def _toml_string(value: object) -> str:
    return json.dumps(str(value), ensure_ascii=True)


def _toml_array(values: list[str], indent: str = "  ") -> str:
    if not values:
        return "[]"
    lines = ["["]
    for value in values:
        lines.append(f"{indent}{_toml_string(value)},")
    lines.append("]")
    return "\n".join(lines)

def _get_environment_section(config: Config) -> str:
    now = datetime.now()
    os_info = f"{platform.system()} {platform.release()}"

    read_paths = [str(p) for p in (config.read_paths or [])] or ["<server-controlled>"]
    sandbox = str(config.sandbox_dir)

    return "\n".join(
        [
            "[environment]",
            f"current_date = {_toml_string(now.strftime('%A, %B %d, %Y'))}",
            f"operating_system = {_toml_string(os_info)}",
            f"working_directory = {_toml_string(config.cwd)}",
            f"sandbox_dir = {_toml_string(sandbox)}",
            f"read_paths = {_toml_array(read_paths)}",
            'tool_access = "Read tools can touch read_paths; write tools are limited to sandbox_dir."',
        ]
    )

def _get_tool_guidelines_section(tools: list[Tool]) -> str:
    regular_tools = [t for t in tools if not t.name.startswith("subagent_")]
    subagent_tools = [t for t in tools if t.name.startswith("subagent_")]

    lines = [
        "[tools]",
        'availability = "Use only tools listed in this section."',
    ]

    for tool in regular_tools:
        description = _moderation_safe_tool_description(tool.description)
        if len(description) > 240:
            description = description[:240] + "..."
        lines.extend(
            [
                "",
                "[[tools.available]]",
                f"name = {_toml_string(tool.name)}",
                f"description = {_toml_string(description)}",
            ]
        )

    for tool in subagent_tools:
        description = _moderation_safe_tool_description(tool.description)
        if len(description) > 240:
            description = description[:240] + "..."
        lines.extend(
            [
                "",
                "[[tools.subagents]]",
                f"name = {_toml_string(tool.name)}",
                f"description = {_toml_string(description)}",
            ]
        )

    best_practices = [
        "Use inspect_input to understand a workbook's sheets or a file's structure before analyzing it.",
        "Use validate_config (with the QA or COMP control type) to validate a config workbook by sheet.",
        "Use analyze_transforms to read and explain FieldOperations transforms.",
        "Use profile_data for statistics, distributions, and data-quality checks on input data.",
        "Use list_dir and glob to locate config workbooks and data files within allow-listed folders.",
        "Use grep to find values or fields by content.",
        "Use read_file to read data, but prefer inspect_input/validate_config/profile_data for workbooks.",
        "Use write_file and generate_report only inside the sandbox_dir.",
        "Use think before acting on complex multi-step tasks.",
        "Use todos to track progress on multi-step work.",
        "Use memory for durable user preferences or notes that should persist across sessions.",
    ]
    if subagent_tools:
        best_practices.append(
            "Use sub-agents (config_inspector, validation_analyst, transform_analyst, "
            "completeness_reconciler, report_writer) for focused, isolated work; pass them precise goals."
        )

    lines.extend(
        [
            "",
            "[tools.best_practices]",
            f"items = {_toml_array(best_practices)}",
        ]
    )

    return "\n".join(lines)


def _moderation_safe_tool_description(description: str) -> str:
    sanitized = description
    for old, new in _TOOL_DESCRIPTION_REPLACEMENTS.items():
        sanitized = sanitized.replace(old, new)
        sanitized = sanitized.replace(old.capitalize(), new.capitalize())
    return sanitized
