"""GET /api/bca/health and GET /api/bca/models."""

from __future__ import annotations
from fastapi import APIRouter

from ai_coding_agent.server.routes._constants import API_PREFIX
from ai_coding_agent.server.deps import get_default_config, get_registry, get_subagent_names
from ai_coding_agent.server.schemas import HealthResponse
from ai_coding_agent.utils.env import (
    get_default_model,
    get_input_folders,
    list_user_sandboxes,
)

router = APIRouter(prefix=API_PREFIX, tags=["health"])


@router.get("/health", response_model=HealthResponse)
async def health() -> HealthResponse:
    cfg = get_default_config()
    return HealthResponse(
        mode="business-config-agent",
        model=cfg.model_name,
        tools=len(get_registry().get_tools()),
        sub_agents=len(get_subagent_names()),
        input_folders=get_input_folders(),
        user_sandboxes=list_user_sandboxes(),
    )


@router.get("/models")
async def models() -> dict:
    default = get_default_model()
    return {
        "models": [
            {"id": default, "name": default, "default": True},
            {"id": "openai/gpt-4o", "name": "GPT-4o"},
            {"id": "anthropic/claude-sonnet-4-6", "name": "Claude Sonnet 4.6"},
            {"id": "mistralai/devstral-2512:free", "name": "Devstral 2512 (free)"},
        ]
    }
