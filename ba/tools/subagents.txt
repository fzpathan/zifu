import asyncio
from typing import Any
from ai_coding_agent.config.config import Config
from ai_coding_agent.tools.base import Tool, ToolInvocation, ToolResult
from dataclasses import dataclass
from pydantic import BaseModel, Field


class SubagentParams(BaseModel):
    goal: str = Field(
        ..., description="The specific task or goal for the sub-agent to accomplish"
    )


@dataclass
class SubagentDefinition:
    name: str
    description: str
    goal_prompt: str
    allowed_tools: list[str] | None = None
    max_turns: int = 20
    timeout_seconds: float = 600


class SubagentTool(Tool):
    def __init__(self, config: Config, definition: SubagentDefinition, *, mutating: bool = True):
        super().__init__(config)
        self.definition = definition
        self._mutating = mutating

    @property
    def name(self) -> str:
        return f"subagent_{self.definition.name}"

    @property
    def description(self) -> str:
        return f"subagent_{self.definition.description}"

    schema = SubagentParams

    def is_mutating(self, params: dict[str, Any]) -> bool:
        return self._mutating

    async def execute(self, invocation: ToolInvocation) -> ToolResult:
        from ai_coding_agent.agent.agent import Agent
        from ai_coding_agent.agent.events import AgentEventType

        params = SubagentParams(**invocation.params)
        if not params.goal:
            return ToolResult.error_result("No goal specified for sub-agent")

        config_dict = self.config.to_dict()
        config_dict["max_turns"] = self.definition.max_turns
        if self.definition.allowed_tools:
            config_dict["allowed_tools"] = self.definition.allowed_tools

        subagent_config = Config(**config_dict)

        prompt = f"""You are a specialized sub-agent with a single, narrow task to complete.

{self.definition.goal_prompt}

YOUR TASK:
{params.goal}

IMPORTANT:
- Focus only on completing the specified task.
- Do not engage in unrelated actions.
- Stay within the configured tool set; do not call tools that are not listed.
- When you have the answer, return a single ```json fenced block that matches the output contract, followed by one short prose paragraph.
- Be concise and direct.
"""

        tool_calls: list[str] = []
        final_response: str | None = None
        error: str | None = None
        terminate_response = "goal"

        try:
            async with Agent(subagent_config) as agent:
                deadline = (
                    asyncio.get_running_loop().time() + self.definition.timeout_seconds
                )

                async for event in agent.run(prompt):
                    if asyncio.get_running_loop().time() > deadline:
                        terminate_response = "timeout"
                        final_response = "Sub-agent timed out"
                        break

                    if event.type == AgentEventType.TOOL_CALL_START:
                        tool_calls.append(event.data.get("name"))
                    elif event.type == AgentEventType.TEXT_COMPLETE:
                        final_response = event.data.get("content")
                    elif event.type == AgentEventType.AGENT_END:
                        if final_response is None:
                            final_response = event.data.get("response")
                    elif event.type == AgentEventType.AGENT_ERROR:
                        terminate_response = "error"
                        error = event.data.get("error", "Unknown")
                        final_response = f"Sub-agent error: {error}"
                        break
        except Exception as e:
            terminate_response = "error"
            error = str(e)
            final_response = f"Sub-agent failed: {e}"

        result = f"""Sub-agent '{self.definition.name}' completed.
Termination: {terminate_response}
Tools called: {', '.join(tool_calls) if tool_calls else 'None'}

Result:
{final_response or 'No response'}
"""

        if error:
            return ToolResult.error_result(result)

        return ToolResult.success_result(result)


# ── Built-in BA sub-agents ────────────────────────────────────────────────────

CONFIG_INSPECTOR = SubagentDefinition(
    name="config_inspector",
    description=(
        "Find ControlDash QA/COMP config workbooks in the input folders and catalog "
        "their sheets and structure (InputFiles, rules, Enrichment, FieldOperations, "
        "OutputRule) plus any obvious structural issues."
    ),
    goal_prompt=(
        "You are the ControlDash config-inspector. Your single mission is to locate "
        "QA/COMP config workbook(s) and catalog their sheets and structure as a JSON "
        "object.\n\n"
        "EXPECTED INPUTS: a folder or a list of workbook paths (relative to the active input folder).\n"
        "EXPECTED OUTPUT: a single ```json fenced block with the contract below, "
        "followed by one short prose paragraph (2-4 sentences) summarising what was found.\n\n"
        "OUTPUT CONTRACT (return as JSON):\n"
        "{\n"
        '  "workbooks": [\n'
        "    {\n"
        '      "file": "<filename>",\n'
        '      "likely_control_type": "QA|COMP|unknown",\n'
        '      "sheets": ["<sheet name>", ...],\n'
        '      "recognized_sheets": ["InputFiles", "rules", "Enrichment", "FieldOperations", "OutputRule"],\n'
        '      "unknown_sheets": ["<sheet name>", ...],\n'
        '      "field_operations_rows": <int|null>,\n'
        '      "issues": ["<string>", ...]\n'
        "    }\n"
        "  ]\n"
        "}\n\n"
        "NEGATIVE BEHAVIOURS:\n"
        "- Do not call other sub-agents.\n"
        "- Do not modify any files; inspection is read-only.\n"
        "- Do not validate or run transforms; just catalog structure.\n"
        "- Do not invent sheets or columns; if a workbook cannot be read, put it in `issues`.\n\n"
        "STYLE: cite file and sheet names verbatim; one paragraph per workbook if helpful."
    ),
    allowed_tools=["read_file", "grep", "glob", "list_dir", "inspect_input"],
    max_turns=12,
    timeout_seconds=180,
)

_BUILTIN_SUBAGENTS: list[SubagentDefinition] = [
    CONFIG_INSPECTOR,
    VALIDATION_ANALYST,
    TRANSFORM_ANALYST,
    COMPLETENESS_RECONCILER,
    REPORT_WRITER,
]

# Per-subagent mutability flag. Read-only sub-agents (config_inspector,
# validation_analyst, transform_analyst, completeness_reconciler) are
# marked non-mutating so the model can fan out inspections in parallel
# rather than serialising them. report_writer writes to the user sandbox
# and stays mutating.
_SUBAGENT_MUTATING: dict[str, bool] = {
    CONFIG_INSPECTOR.name: False,
    VALIDATION_ANALYST.name: False,
    TRANSFORM_ANALYST.name: False,
    COMPLETENESS_RECONCILER.name: False,
    REPORT_WRITER.name: True,
}

_USER_SUBAGENTS: list[SubagentDefinition] = []
_USER_SUBAGENT_MUTATING: dict[str, bool] = {}


def register_subagent(definition: SubagentDefinition, *, mutating: bool = True) -> None:
    """Add a new sub-agent to the global registry. Callable at app startup.

    Example
    -------
    >>> from ai_coding_agent.tools.subagents import (
    ...     register_subagent, SubagentDefinition,
    ... )
    >>> register_subagent(SubagentDefinition(
    ...     name="industry_analyst",
    ...     description="Run industry-specific checks against a CSV.",
    ...     goal_prompt="...",
    ...     allowed_tools=["read_file", "profile_data"],
    ... ), mutating=False)
    """
    _USER_SUBAGENTS.append(definition)
    _USER_SUBAGENT_MUTATING[definition.name] = mutating


def get_subagent_definitions(
    extra: list[SubagentDefinition] | None = None,
) -> list[SubagentDefinition]:
    """Built-ins + globally registered + caller-supplied (one-shot) extras."""
    return list(_BUILTIN_SUBAGENTS) + list(_USER_SUBAGENTS) + (extra or [])


def get_subagent_mutating_map() -> dict[str, bool]:
    """Return ``{name: is_mutating}`` for every registered sub-agent."""
    return {**_SUBAGENT_MUTATING, **_USER_SUBAGENT_MUTATING}
