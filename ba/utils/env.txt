"""Environment / config loading for the BA server.

Environment variables (all optional unless noted):

* ``API_KEY``              — required, OpenAI-compatible key.
* ``BASE_URL``             — required, OpenAI-compatible base URL.
* ``BA_INPUT_ROOT``        — absolute path to the shared inputs folder.
  Defaults to ``<cwd>/data/inputs`` if unset, so the server can boot
  without configuration.
* ``BA_OUTPUT_ROOT``       — absolute path where per-user output sandboxes
  are created (``<BA_OUTPUT_ROOT>/<user_id>``). Defaults to
  ``<cwd>/data/outputs``.
* ``BA_INPUT_FOLDERS``     — optional, comma-separated list of immediate
  subfolders of ``BA_INPUT_ROOT`` that should be exposed. If unset, all
  immediate subfolders are exposed.
* ``MODEL_NAME``           — optional, default model. Defaults to
  ``gpt-4o-mini``.
* ``BA_DEFAULT_USER``      — optional, default user_id. Defaults to
  ``demo-user``.
* ``HOST`` / ``PORT``      — optional, uvicorn bind address.
"""

from __future__ import annotations
import os
import threading
import time
from pathlib import Path
from typing import Iterable

try:
    from dotenv import find_dotenv, load_dotenv as _load_dotenv
    # Auto-load a .env from the cwd (or its parents) at import time.
    # Shell-set environment variables take precedence over .env values.
    _dotenv_path = find_dotenv(".env", usecwd=True)
    if _dotenv_path:
        _load_dotenv(_dotenv_path, override=False)
except ImportError:  # python-dotenv not installed; .env loading is a no-op
    pass


def get_input_root() -> Path:
    value = os.environ.get("BA_INPUT_ROOT", "").strip()
    if not value:
        # Always-on default so the server can start without env vars.
        value = str(Path.cwd() / "data" / "inputs")
        os.environ["BA_INPUT_ROOT"] = value
    return Path(value).resolve()


def get_output_root() -> Path:
    value = os.environ.get("BA_OUTPUT_ROOT", "").strip()
    if not value:
        # Always-on default so the server can start without env vars.
        value = str(Path.cwd() / "data" / "outputs")
        os.environ["BA_OUTPUT_ROOT"] = value
    return Path(value).resolve()


def get_input_folders() -> list[str]:
    """Names of the immediate subfolders of BA_INPUT_ROOT that should be exposed."""
    explicit = os.environ.get("BA_INPUT_FOLDERS", "").strip()
    if explicit:
        return [name.strip() for name in explicit.split(",") if name.strip()]
    root = get_input_root()
    if not root.exists():
        return []
    return sorted(
        p.name for p in root.iterdir() if p.is_dir() and not p.name.startswith(".")
    )


def get_user_sandbox_dir(user_id: str) -> Path:
    if not user_id or any(c in user_id for c in r'/\:*?"<>|'):
        raise ValueError(f"invalid user_id: {user_id!r}")
    sandbox = (get_output_root() / user_id).resolve()
    sandbox.mkdir(parents=True, exist_ok=True)
    return sandbox


def get_default_model() -> str:
    return os.environ.get("MODEL_NAME", "openai/gpt-4o-mini")


def get_default_user() -> str:
    return os.environ.get("BA_DEFAULT_USER", "demo-user")


def get_host() -> str:
    return os.environ.get("HOST", "127.0.0.1")


def get_port() -> int:
    try:
        return int(os.environ.get("PORT", "8000"))
    except ValueError:
        return 8000


def ensure_roots_exist() -> None:
    """Create BA_INPUT_ROOT and BA_OUTPUT_ROOT if they are missing.

    The always-on default paths (./data/inputs and ./data/outputs) are
    created on first use, so the server can boot without any environment
    configuration.
    """
    root = get_input_root()
    root.mkdir(parents=True, exist_ok=True)
    out = get_output_root()
    out.mkdir(parents=True, exist_ok=True)


def list_user_sandboxes() -> list[str]:
    try:
        root = get_output_root()
    except RuntimeError:
        return []
    if not root.exists():
        return []
    return sorted(p.name for p in root.iterdir() if p.is_dir())


# TTL cache for list_user_artifacts. The function walks the user sandbox
# with rglob on every call, which is fine for small sandboxes but adds up
# when the frontend polls /api/bca/artifacts after each chat turn.
# Cache tuple: (monotonic_ts, wall_ts, items)
#   monotonic_ts — used for TTL check (time.monotonic(), stable interval)
#   wall_ts      — used for st_mtime comparison (time.time(), Unix epoch)
_ARTIFACT_CACHE_TTL_SEC = 5.0
_artifact_cache: dict[str, tuple[float, float, list[dict]]] = {}
_artifact_lock = threading.Lock()


def _walk_sandbox_artifacts(sandbox: Path) -> list[dict]:
    items: list[dict] = []
    for p in sandbox.rglob("*"):
        if not p.is_file():
            continue
        try:
            stat = p.stat()
        except OSError:
            continue
        items.append(
            {
                "path": str(p.relative_to(sandbox)).replace("\\", "/"),
                "size": stat.st_size,
                "modified": stat.st_mtime,
            }
        )
    items.sort(key=lambda d: d["modified"], reverse=True)
    return items


def list_user_artifacts(user_id: str) -> list[dict]:
    sandbox = get_user_sandbox_dir(user_id)
    now_mono = time.monotonic()
    with _artifact_lock:
        cached = _artifact_cache.get(user_id)
        if cached is not None:
            ts_mono, ts_wall, items = cached
            if now_mono - ts_mono < _ARTIFACT_CACHE_TTL_SEC:
                return items
            # TTL expired — reuse if the sandbox directory itself hasn't
            # changed. Compare st_mtime (Unix epoch) against ts_wall (also
            # Unix epoch) so the clocks are compatible.
            try:
                sandbox_mtime = sandbox.stat().st_mtime
            except OSError:
                sandbox_mtime = None
            if sandbox_mtime is not None and sandbox_mtime <= ts_wall:
                _artifact_cache[user_id] = (now_mono, ts_wall, items)
                return items

    items = _walk_sandbox_artifacts(sandbox)
    with _artifact_lock:
        _artifact_cache[user_id] = (time.monotonic(), time.time(), items)
    return items


def invalidate_artifact_cache(user_id: str | None = None) -> None:
    """Drop cached artifact listings. Pass ``user_id`` for one user or
    ``None`` to clear the whole cache (e.g. after a write)."""
    with _artifact_lock:
        if user_id is None:
            _artifact_cache.clear()
        else:
            _artifact_cache.pop(user_id, None)
