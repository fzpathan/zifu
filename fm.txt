export const EMPTY_SUPERVISORY_FILTERS = {
    regulation: [],
    asset_class: [],
    control_type: [],
    data_type: [],
    sub_control_type: [],
    remediation_status: []
};

export const sanitizeFilters = (filters, emptyFilters = EMPTY_SUPERVISORY_FILTERS) => {
    const next = { ...emptyFilters, ...(filters || {}) };
    Object.keys(next).forEach((key) => {
        if (!Array.isArray(next[key])) {
            next[key] = [];
        }
    });
    return next;
};

export const filtersEqual = (left, right) => {
    const keys = Object.keys(EMPTY_SUPERVISORY_FILTERS);
    return keys.every((key) => {
        const leftValues = left?.[key] || [];
        const rightValues = right?.[key] || [];
        if (leftValues.length !== rightValues.length) return false;
        return leftValues.every((value, index) => value === rightValues[index]);
    });
};

export const STARTUP_FILTER_KEYS = ['regulation', 'asset_class', 'data_type'];

export const buildInitialFilters = (response) => {
    const nextFilters = { ...EMPTY_SUPERVISORY_FILTERS };
    const configuredDefaults = response?.default_initial_filters || {};
    const filterOptions = response?.filter_options || {};

    STARTUP_FILTER_KEYS.forEach((key) => {
        const availableOptions = filterOptions[key] || [];
        const validConfiguredDefaults = (configuredDefaults[key] || []).filter((value) => availableOptions.includes(value));
        if (validConfiguredDefaults.length > 0) {
            nextFilters[key] = validConfiguredDefaults;
        } else if (availableOptions.length > 0) {
            nextFilters[key] = [availableOptions[0]];
        }
    });

    return sanitizeFilters(nextFilters);
};

export const hasAnyFilterValues = (filters) =>
    Object.values(filters || {}).some((values) => Array.isArray(values) && values.length > 0);

const getTodayDate = () => {
    const today = new Date();
    const year = today.getFullYear();
    const month = String(today.getMonth() + 1).padStart(2, '0');
    const day = String(today.getDate()).padStart(2, '0');
    return `${year}-${month}-${day}`;
};

const matchesHierarchyFilters = (file, activeFilters, excludeKey = null) => {
    const checks = [
        ['regulation', activeFilters.regulation],
        ['asset_class', activeFilters.asset_class],
        ['sub_control_name', activeFilters.sub_control_name],
        ['control_name', activeFilters.control_name],
    ];

    return checks.every(([key, value]) => {
        if (key === excludeKey || !value) return true;
        return file[key] === value;
    });
};

const matchesSidebarFilters = (file, activeFilters) => {
    if (!matchesHierarchyFilters(file, activeFilters)) return false;
    if (activeFilters.frequency && file.frequency !== activeFilters.frequency) return false;
    if (activeFilters.status &&
        (file.status || '').toLowerCase() !== activeFilters.status.toLowerCase()) {
        return false;
    }
    return true;
};

const uniqueSortedValues = (items, selector) => (
    [...new Set(items.map(selector).filter(Boolean))].sort()
);

const matchesColumnFilters = (file, columnFilters) => {
    if (columnFilters.display_name) {
        const search = columnFilters.display_name.toLowerCase();
        const displayName = (file.display_name || '').toLowerCase();
        const fileName = (file.file_name || '').toLowerCase();
        if (!displayName.includes(search) && !fileName.includes(search)) return false;
    }
    if (columnFilters.regulation) {
        const search = columnFilters.regulation.toLowerCase();
        if (!(file.regulation || '').toLowerCase().includes(search)) return false;
    }
    if (columnFilters.asset_class) {
        const search = columnFilters.asset_class.toLowerCase();
        if (!(file.asset_class || '').toLowerCase().includes(search)) return false;
    }
    if (columnFilters.sub_control_name) {
        const search = columnFilters.sub_control_name.toLowerCase();
        if (!(file.sub_control_name || '').toLowerCase().includes(search)) return false;
    }
    if (columnFilters.control_name) {
        const search = columnFilters.control_name.toLowerCase();
        if (!(file.control_name || '').toLowerCase().includes(search)) return false;
    }
    if (columnFilters.frequency) {
        const search = columnFilters.frequency.toLowerCase();
        if (!(file.frequency || '').toLowerCase().includes(search)) return false;
    }
    if (columnFilters.status &&
        (file.status || '').toLowerCase() !== columnFilters.status.toLowerCase()) {
        return false;
    }
    if (columnFilters.note &&
        !(file.note || '').toLowerCase().includes(columnFilters.note.toLowerCase())) {
        return false;
    }

    return true;
};

useSupervisoryData dashboard
import { useState, useEffect, useCallback, useMemo, useRef } from 'react';
import { ApiService } from '../../../services/api';
import { BUCKET_SET_CONFIG, DEFAULT_BUCKET_SET_OPTIONS, DEFAULT_AGE_BUCKETS, buildBucketSummary } from '../utils/bucketConfig';
import { AVAILABLE_COLUMNS } from '../utils/columnConfig';
import { EMPTY_SUPERVISORY_FILTERS, buildInitialFilters, filtersEqual, hasAnyFilterValues, sanitizeFilters } from '../utils/filterStateUtils';
import { cacheGet, cacheSet } from './useLocalCache';

// Cache keys
const CK_FILTER_OPTIONS = 'filterOptions';
const CK_AGGREGATIONS = 'aggregations';
const LS_SELECTED_COLUMNS = 'supervisory_selected_columns';

export const useSupervisoryData = (currentUser) => {
    const cachedFilterOptions = cacheGet(CK_FILTER_OPTIONS);
    const cachedInitialFilters = cachedFilterOptions ? buildInitialFilters(cachedFilterOptions) : { ...EMPTY_SUPERVISORY_FILTERS };

    // State
    const [filterOptions, setFilterOptions] = useState(() =>
        cachedFilterOptions?.filter_options ?? {
            regulation: [], asset_class: [], control_type: [],
            data_type: [], sub_control_type: [], remediation_status: []
        }
    );

    const [savedFilters, setSavedFilters] = useState([]);
    const [selectedSavedFilterId, setSelectedSavedFilterId] = useState('');
    const [savedFilterName, setSavedFilterName] = useState('');
    const [isSavingFilter, setIsSavingFilter] = useState(false);
    const [selectedBucketSet, setSelectedBucketSet] = useState(() =>
        cachedFilterOptions?.default_bucket_set ?? 'CFTC'
    );
    const [availableBucketSets, setAvailableBucketSets] = useState(() =>
        cachedFilterOptions?.bucket_sets ?? DEFAULT_BUCKET_SET_OPTIONS
    );
    const [activeAgeBuckets, setActiveAgeBuckets] = useState(() => {
        if (cachedFilterOptions) {
            const bs = cachedFilterOptions.default_bucket_set || 'CFTC';
            return cachedFilterOptions.age_buckets_by_set?.[bs] || cachedFilterOptions.age_buckets || DEFAULT_AGE_BUCKETS;
        }
        return DEFAULT_AGE_BUCKETS;
    });
    const [selectedColumns, setSelectedColumns] = useState(() => {
        try {
            const stored = window.localStorage.getItem(LS_SELECTED_COLUMNS);
            if (stored) {
                const fields = JSON.parse(stored);
                if (Array.isArray(fields) && fields.length > 0) {
                    const cols = fields
                        .map(f => AVAILABLE_COLUMNS.find(c => c.field === f))
                        .filter(Boolean);
                    if (cols.length > 0) return cols;
                }
            }
        } catch { /* ignore */ }
        return [
            AVAILABLE_COLUMNS[0], AVAILABLE_COLUMNS[1], AVAILABLE_COLUMNS[2],
            AVAILABLE_COLUMNS[6], AVAILABLE_COLUMNS[7], AVAILABLE_COLUMNS[8]
        ];
    });

    const [filters, setFilters] = useState(cachedInitialFilters);
    const [draftFilters, setDraftFilters] = useState(cachedInitialFilters);
    const [draftBucketSet, setDraftBucketSet] = useState(() =>
        cachedFilterOptions?.default_bucket_set ?? 'CFTC'
    );

    // Try to seed aggregation state from cache
    const cachedAgg = cacheGet(CK_AGGREGATIONS);
    const [aggregations, setAggregations] = useState(cachedAgg?.aggregations ?? []);
    const [summary, setSummary] = useState(cachedAgg?.summary ?? {
        total_count: 0, unremediated_count: 0,
        total: buildBucketSummary(DEFAULT_AGE_BUCKETS),
        unremediated: buildBucketSummary(DEFAULT_AGE_BUCKETS)
    });
    const [totalRecords, setTotalRecords] = useState(cachedAgg?.total_records ?? 0);
    const [filteredRecords, setFilteredRecords] = useState(cachedAgg?.filtered_records ?? 0);

    // Loading starts as false if we have cached data, true otherwise
    const hasCachedAggregations = useRef(!!cachedAgg);
    const [isLoading, setIsLoading] = useState(!hasCachedAggregations.current);
    const [isInitialFiltersReady, setIsInitialFiltersReady] = useState(false);
    const [error, setError] = useState(null);

    const initialLoadDone = useRef(false);
    const skipGroupByEffectRef = useRef(false);
    const prevGroupByKeyRef = useRef(null);
    const appliedQueryRef = useRef({ filters: cachedInitialFilters, selectedBucketSet: cachedFilterOptions?.default_bucket_set ?? 'CFTC', groupBy: [] });

    // Derived State
    const groupBy = useMemo(() => selectedColumns.map(c => c.field), [selectedColumns]);
    appliedQueryRef.current = { filters, selectedBucketSet, groupBy };

    // Persist selectedColumns to localStorage on every change
    useEffect(() => {
        try {
            const fields = selectedColumns.map(c => c.field);
            window.localStorage.setItem(LS_SELECTED_COLUMNS, JSON.stringify(fields));
        } catch { /* ignore */ }
    }, [selectedColumns]);
    const hasActiveFilters = useMemo(
        () => hasAnyFilterValues(draftFilters),
        [draftFilters]
    );

    const hasPendingFilterChanges = useMemo(
        () => !filtersEqual(draftFilters, filters) || draftBucketSet !== selectedBucketSet,
        [draftFilters, filters, draftBucketSet, selectedBucketSet]
    );

    // ─── Helper: apply filter options response to state ──────────

    const applyFilterOptionsResponse = useCallback((response) => {
        setFilterOptions(response.filter_options);
        setTotalRecords(response.total_records);
        const bucketSets = Array.isArray(response.bucket_sets) && response.bucket_sets.length > 0
            ? response.bucket_sets
            : DEFAULT_BUCKET_SET_OPTIONS;
        setAvailableBucketSets(bucketSets);

        const defaultBucketSet = bucketSets.includes(response.default_bucket_set)
            ? response.default_bucket_set
            : bucketSets[0];

        setSelectedBucketSet(defaultBucketSet);
        setActiveAgeBuckets(
            response.age_buckets_by_set?.[defaultBucketSet] ||
            response.age_buckets ||
            BUCKET_SET_CONFIG[defaultBucketSet]?.buckets ||
            DEFAULT_AGE_BUCKETS
        );

        const sanitizedFilters = buildInitialFilters(response);
        setFilters(sanitizedFilters);
        setDraftFilters(sanitizedFilters);
        setDraftBucketSet(defaultBucketSet);
    }, []);

    // ─── Helper: apply aggregation response to state ─────────────

    const applyAggregationResponse = useCallback((response) => {
        const activeBucketSet = response.bucket_set;

        if (activeBucketSet) {
            setSelectedBucketSet(activeBucketSet);
            setDraftBucketSet(activeBucketSet);
            setActiveAgeBuckets(
                response.age_buckets ||
                BUCKET_SET_CONFIG[activeBucketSet]?.buckets ||
                DEFAULT_AGE_BUCKETS
            );
        } else {
            // Fallback if backend doesn't provide it (shouldn't happen with our fix)
            setActiveAgeBuckets(
                response.age_buckets ||
                DEFAULT_AGE_BUCKETS
            );
        }

        setAggregations(response.aggregations || []);
        setSummary(response.summary);
        setFilteredRecords(response.filtered_records);
        setTotalRecords(response.total_records);
    }, []); // Stable: no dependencies on external state variables

    // ─── Separate filter options load (fallback & refresh) ───────

    const loadFilterOptions = useCallback(async () => {
        try {
            const response = await ApiService.getSupervisoryFilterOptions();
            if (response.success) {
                cacheSet(CK_FILTER_OPTIONS, response);
                applyFilterOptionsResponse(response);
            }
        } catch (err) {
            setError('Failed to load filter options');
        } finally {
            setIsInitialFiltersReady(true);
        }
    }, [applyFilterOptionsResponse]);

    // ─── Aggregation load (for filter/groupBy changes) ───────────

    const fetchAggregations = useCallback(async (nextFilters, nextBucketSet, nextGroupBy = groupBy) => {
        setIsLoading(true);
        setError(null);
        try {
            const response = await ApiService.getSupervisoryAggregations(nextFilters, nextGroupBy, nextBucketSet);
            if (response.success) {
                cacheSet(CK_AGGREGATIONS, response);
                applyAggregationResponse(response);
            }
        } catch (err) {
            setError(err.message || 'Failed to load aggregations');
        } finally {
            setIsLoading(false);
        }
    }, [groupBy, applyAggregationResponse]);

    const loadAggregations = useCallback(async () => {
        await fetchAggregations(filters, selectedBucketSet, groupBy);
    }, [fetchAggregations, filters, selectedBucketSet, groupBy]);

    // ─── Combined initial load (single round-trip) ───────────────

    const loadInitial = useCallback(async () => {
        try {
            const response = await ApiService.getSupervisoryInitialLoad();
            if (response.success) {
                cacheSet(CK_FILTER_OPTIONS, response);

                applyFilterOptionsResponse(response);

                if (response.default_aggregations?.success) {
                    cacheSet(CK_AGGREGATIONS, response.default_aggregations);
                    applyAggregationResponse(response.default_aggregations);
                }

                prevGroupByKeyRef.current = selectedColumns.map((column) => column.field).join('|');
                initialLoadDone.current = true;
            }
        } catch (err) {
            setError('Failed to load initial data');
            try {
                await loadFilterOptions();
                const cached = cacheGet(CK_FILTER_OPTIONS);
                if (cached) {
                    const nextFilters = buildInitialFilters(cached);
                    setFilters(nextFilters);
                    setDraftFilters(nextFilters);
                    prevGroupByKeyRef.current = selectedColumns.map((column) => column.field).join('|');
                    initialLoadDone.current = true;
                    await fetchAggregations(nextFilters, cached.default_bucket_set || selectedBucketSet, selectedColumns.map((column) => column.field));
                }
            } catch {
                // already handled
            }
        } finally {
            setIsInitialFiltersReady(true);
            setIsLoading(false);
        }
    }, [applyFilterOptionsResponse, applyAggregationResponse, loadFilterOptions, fetchAggregations, selectedBucketSet, selectedColumns]);

    const loadSavedFilters = useCallback(async () => {
        try {
            const response = await ApiService.getSupervisorySavedFilters();
            if (response.success) {
                setSavedFilters(response.filters || []);
            }
        } catch (err) {
            console.error('Failed to load saved filters', err);
        }
    }, []);

    // ─── Force Refresh (bypass cache, re-fetch everything) ───────

    const forceRefresh = useCallback(() => {
        loadFilterOptions();
        loadAggregations();
    }, [loadFilterOptions, loadAggregations]);

    // ─── Effects ─────────────────────────────────────────────────

    // On mount: use combined initial-load endpoint (1 round-trip)
    useEffect(() => {
        loadInitial();
        loadSavedFilters();
    }, [loadInitial, loadSavedFilters]);

    const groupByKey = groupBy.join('|');

    // When groupBy changes AFTER initial load → re-fetch aggregations with applied filters
    useEffect(() => {
        if (skipGroupByEffectRef.current) return;
        if (!initialLoadDone.current || groupBy.length === 0) return;

        const groupByChanged = prevGroupByKeyRef.current !== groupByKey;
        prevGroupByKeyRef.current = groupByKey;
        if (!groupByChanged) return;

        const { filters: appliedFilters, selectedBucketSet: appliedBucketSet, groupBy: appliedGroupBy } = appliedQueryRef.current;
        fetchAggregations(appliedFilters, appliedBucketSet, appliedGroupBy);
    }, [groupByKey, fetchAggregations, groupBy.length]);

    const handleFilterChange = (filterKey, values) => {
        setDraftFilters(prev => ({ ...prev, [filterKey]: values }));
    };

    const handleDraftBucketSetChange = (bucketSet) => {
        setDraftBucketSet(bucketSet);
    };

    const applyFilters = useCallback(async () => {
        const nextFilters = sanitizeFilters(draftFilters);
        const nextBucketSet = draftBucketSet;
        initialLoadDone.current = true;
        setFilters(nextFilters);
        setSelectedBucketSet(nextBucketSet);
        await fetchAggregations(nextFilters, nextBucketSet, groupBy);
    }, [draftFilters, draftBucketSet, fetchAggregations, groupBy]);

    const clearAllFilters = () => {
        setDraftFilters({ ...EMPTY_SUPERVISORY_FILTERS });
    };

    const applySavedFilter = useCallback(async (saved) => {
        if (!saved) return;
        initialLoadDone.current = true;
        const nextFilters = sanitizeFilters(saved.filters);

        setFilters(nextFilters);
        setDraftFilters(nextFilters);

        let nextGroupBy = groupBy;
        if (Array.isArray(saved.group_by) && saved.group_by.length > 0) {
            const nextSelected = saved.group_by
                .map(field => AVAILABLE_COLUMNS.find(col => col.field === field))
                .filter(Boolean);
            if (nextSelected.length > 0) {
                skipGroupByEffectRef.current = true;
                setSelectedColumns(nextSelected);
                nextGroupBy = nextSelected.map(col => col.field);
            }
        }

        await fetchAggregations(nextFilters, selectedBucketSet, nextGroupBy);
        prevGroupByKeyRef.current = nextGroupBy.join('|');
        skipGroupByEffectRef.current = false;
    }, [fetchAggregations, groupBy, selectedBucketSet]);

    const handleSavedFilterSelect = (e) => {
        const value = e.target.value;
        setSelectedSavedFilterId(value);
        const selected = savedFilters.find(f => f.id === value);
        applySavedFilter(selected);
    };

    const handleSaveCurrentFilter = async () => {
        if (!savedFilterName.trim()) return;
        setIsSavingFilter(true);
        try {
            const payload = {
                name: savedFilterName.trim(),
                filters: sanitizeFilters(draftFilters),
                group_by: selectedColumns.map(col => col.field),
                created_by: currentUser?.id || currentUser?.username || 'anonymous',
                created_by_name: currentUser?.name || currentUser?.username || 'anonymous'
            };
            const response = await ApiService.saveSupervisoryFilter(payload);
            if (response.success) {
                setSavedFilters(prev => [response.filter, ...prev]);
                setSavedFilterName('');
                setSelectedSavedFilterId(response.filter.id);
            }
        } catch (err) {
            console.error('Failed to save filter', err);
        } finally {
            setIsSavingFilter(false);
        }
    };

    return {
        // State
        filterOptions,
        savedFilters,
        selectedSavedFilterId,
        savedFilterName,
        setSavedFilterName,
        isSavingFilter,
        selectedBucketSet,
        availableBucketSets,
        activeAgeBuckets,
        selectedColumns,
        setSelectedColumns,
        filters,
        draftFilters,
        draftBucketSet,
        aggregations,
        summary,
        totalRecords,
        filteredRecords,
        isLoading,
        error,
        hasActiveFilters,
        hasPendingFilterChanges,

        // Actions
        loadAggregations,
        forceRefresh,
        handleFilterChange,
        handleDraftBucketSetChange,
        applyFilters,
        clearAllFilters,
        handleSavedFilterSelect,
        handleSaveCurrentFilter
    };
};

export const sanitizeFilters = (filters, emptyFilters = EMPTY_SUPERVISORY_FILTERS) => {
    const next = { ...emptyFilters, ...(filters || {}) };
    Object.keys(next).forEach((key) => {
        if (!Array.isArray(next[key])) {
            next[key] = [];
        }
    });
    return next;
};

export const filtersEqual = (left, right) => {
    const keys = Object.keys(EMPTY_SUPERVISORY_FILTERS);
    return keys.every((key) => {
        const leftValues = left?.[key] || [];
        const rightValues = right?.[key] || [];
        if (leftValues.length !== rightValues.length) return false;
        return leftValues.every((value, index) => value === rightValues[index]);
    });
};

export const STARTUP_FILTER_KEYS = ['regulation', 'asset_class', 'data_type'];

export const buildInitialFilters = (response) => {

            nextFilters[key] = validConfiguredDefaults;
        } else if (availableOptions.length > 0) {
            nextFilters[key] = [availableOptions[0]];
        }
    });

    return sanitizeFilters(nextFilters);
};

const PLAN_COLORS = ['#1d4ed8', '#059669', '#d97706', '#7c3aed', '#c2410c', '#be123c', '#0f766e'];

function buildPlanColorMap(series) {
    const plans = [...new Set((series || []).map((item) => item.plan))];
    return Object.fromEntries(plans.map((plan, index) => [plan, PLAN_COLORS[index % PLAN_COLORS.length]]));
}

function withAbbreviatedSeries(series = []) {
    return series.map((item) => ({
        ...item,
        shortLabel: abbreviateRemediationPlan(item.plan || item.label)
    }));
}

function aggregateMonthlyChart(chart, selectedPlans) {
    const series = withAbbreviatedSeries(
        (chart?.series || []).filter((item) => selectedPlans.has(item.plan))
    );
    const planColorMap = buildPlanColorMap(series);
    const byPeriod = new Map();

    (chart?.data || []).forEach((row) => {
        const key = row.period_key;
        if (!byPeriod.has(key)) {
            const base = {
                period_key: row.period_key,
                period_label: row.period_label,
                remediated_total: 0,
                unremediated_total: 0
            };
            series.forEach((item) => {
                base[`remediated__${item.key}`] = 0;
                base[`unremediated__${item.key}`] = 0;
            });
            byPeriod.set(key, base);
        }

        const target = byPeriod.get(key);
        const isRemediated = String(row.status).toLowerCase().includes('remediated') && !String(row.status).toLowerCase().includes('unremediated');
        const totalKey = isRemediated ? 'remediated_total' : 'unremediated_total';
        target[totalKey] += row.total_count || 0;
        series.forEach((item) => {
            const dataKey = `${isRemediated ? 'remediated' : 'unremediated'}__${item.key}`;
            target[dataKey] += row[item.key] || 0;
        });
    });

    const orderedData = Array.from(byPeriod.values()).sort((a, b) => String(a.period_key).localeCompare(String(b.period_key)));

    let cumulativeRemediated = 0;
    let cumulativeUnremediated = 0;
    orderedData.forEach((row) => {
        cumulativeRemediated += row.remediated_total || 0;
        cumulativeUnremediated += row.unremediated_total || 0;
        row.cumulative_remediated_total = cumulativeRemediated;
        row.cumulative_unremediated_total = cumulativeUnremediated;
    });

    return {
        data: orderedData,
        series,
        planColorMap
    };
}

function aggregateDailyStatusChart(chart, targetStatus, selectedPlans, limit = null) {
    const normalizedStatus = targetStatus.toLowerCase();
    const series = withAbbreviatedSeries(
        (chart?.series || []).filter((item) => selectedPlans.has(item.plan))
    );
    const matchingRows = (chart?.data || []).filter((row) => String(row.status).toLowerCase() === normalizedStatus);
    const rowsToUse = limit ? matchingRows.slice(-limit) : matchingRows;
    const planColorMap = buildPlanColorMap(series);

    const data = rowsToUse.map((row) => {
        const nextRow = {
            period_key: row.period_key,
            period_label: row.period_label,
            total_count: row.total_count || 0
        };
        series.forEach((item) => {
            nextRow[item.key] = row[item.key] || 0;
        });
        return nextRow;
    }).sort((a, b) => String(a.period_key).localeCompare(String(b.period_key)));

    let cumulativeTotal = 0;
    data.forEach((row) => {
        cumulativeTotal += row.total_count || 0;
        row.cumulative_total = cumulativeTotal;
    });

    return {
        data,
        series,
        planColorMap,
        statusLabel: targetStatus
    };
}

import { useCallback, useEffect, useMemo, useRef, useState } from 'react';
import { ApiService } from '../../../services/api';
import { cacheGet, cacheSet } from '../../supervisory-dashboard/hooks/useLocalCache';
import { AVAILABLE_COLUMNS } from '../../supervisory-dashboard/utils/columnConfig';
import {
    EMPTY_SUPERVISORY_FILTERS,
    buildInitialFilters,
    filtersEqual,
    hasAnyFilterValues,
    sanitizeFilters
} from '../../supervisory-dashboard/utils/filterStateUtils';

const CK_FILTER_OPTIONS = 'supervisory_trends_v2_filter_options';
const CK_TRENDS = 'supervisory_trends_v2_data';
const LS_SELECTED_COLUMNS = 'supervisory_trends_selected_columns';

export const useSupervisoryTrendData = (currentUser) => {
    const cachedFilters = cacheGet(CK_FILTER_OPTIONS);
    const rawCachedTrends = cacheGet(CK_TRENDS);
    const cachedTrends = rawCachedTrends?.grouped_tables ? rawCachedTrends : null;
    const cachedInitialFilters = cachedFilters ? buildInitialFilters(cachedFilters) : { ...EMPTY_SUPERVISORY_FILTERS };

    const [filterOptions, setFilterOptions] = useState(
        cachedFilters?.filter_options ?? EMPTY_SUPERVISORY_FILTERS
    );
    const [filters, setFilters] = useState(cachedInitialFilters);
    const [draftFilters, setDraftFilters] = useState(cachedInitialFilters);
    const [savedFilters, setSavedFilters] = useState([]);
    const [selectedSavedFilterId, setSelectedSavedFilterId] = useState('');
    const [savedFilterName, setSavedFilterName] = useState('');
    const [isSavingFilter, setIsSavingFilter] = useState(false);
    const [selectedColumns, setSelectedColumns] = useState(() => {
        try {
            const stored = window.localStorage.getItem(LS_SELECTED_COLUMNS);
            if (stored) {
                const fields = JSON.parse(stored);
                if (Array.isArray(fields) && fields.length > 0) {
                    const cols = fields
                        .map((field) => AVAILABLE_COLUMNS.find((column) => column.field === field))
                        .filter(Boolean);
                    if (cols.length > 0) return cols;
                }
            }
        } catch {
            // ignore localStorage parse failures
        }
        return [AVAILABLE_COLUMNS[0], AVAILABLE_COLUMNS[1], AVAILABLE_COLUMNS[2]];
    });
    const [trendData, setTrendData] = useState(cachedTrends ?? null);
    const [totalRecords, setTotalRecords] = useState(cachedFilters?.total_records ?? cachedTrends?.total_records ?? 0);
    const [filteredRecords, setFilteredRecords] = useState(cachedTrends?.filtered_records ?? 0);
    const [isLoading, setIsLoading] = useState(!cachedTrends);
    const [error, setError] = useState(null);

    const initialLoadDone = useRef(false);
    const skipColumnEffectRef = useRef(false);
    const prevColumnKeyRef = useRef(null);
    const appliedQueryRef = useRef({ filters: cachedInitialFilters, selectedColumns: [] });
    appliedQueryRef.current = { filters, selectedColumns };

    const selectedColumnKey = selectedColumns.map((column) => column.field).join('|');

    const hasActiveFilters = useMemo(
        () => hasAnyFilterValues(draftFilters),
        [draftFilters]
    );

    const hasPendingFilterChanges = useMemo(
        () => !filtersEqual(draftFilters, filters),
        [draftFilters, filters]
    );

    const buildInitialFiltersFromResponse = useCallback((response) => buildInitialFilters(response), []);

    const applyFilterOptions = useCallback((response) => {
        setFilterOptions(response.filter_options || EMPTY_SUPERVISORY_FILTERS);
        setTotalRecords(response.total_records || 0);
        cacheSet(CK_FILTER_OPTIONS, response);
    }, []);

    const applyTrendResponse = useCallback((response) => {
        setTrendData(response);
        setFilteredRecords(response.filtered_records || 0);
        setTotalRecords(response.total_records || 0);
        cacheSet(CK_TRENDS, response);
    }, []);

    useEffect(() => {
        try {
            window.localStorage.setItem(
                LS_SELECTED_COLUMNS,
                JSON.stringify(selectedColumns.map((column) => column.field))
            );
        } catch {
            // ignore localStorage write failures
        }
    }, [selectedColumns]);

    const fetchTrends = useCallback(async (nextFilters, nextSelectedColumns = selectedColumns) => {
        setIsLoading(true);
        setError(null);
        try {
            const response = await ApiService.getSupervisoryTrends(
                nextFilters,
                nextSelectedColumns.map((column) => column.field)
            );
            if (response.success) {
                applyTrendResponse(response);
            }
        } catch (err) {
            setError(err.message || 'Failed to load trend analytics');
        } finally {
            setIsLoading(false);
        }
    }, [applyTrendResponse, selectedColumns]);

    const loadSavedFilters = useCallback(async () => {
        try {
            const response = await ApiService.getSupervisorySavedFilters();
            if (response.success) {
                setSavedFilters(response.filters || []);
            }
        } catch (err) {
            console.error('Failed to load saved filters', err);
        }
    }, []);

    const loadInitial = useCallback(async () => {
        setIsLoading(true);
        setError(null);
        try {
            const response = await ApiService.getSupervisoryFilterOptions();
            applyFilterOptions(response);
            const nextFilters = buildInitialFiltersFromResponse(response);
            setFilters(nextFilters);
            setDraftFilters(nextFilters);
            const trendResponse = await ApiService.getSupervisoryTrends(
                nextFilters,
                selectedColumns.map((column) => column.field)
            );
            if (trendResponse.success) {
                applyTrendResponse(trendResponse);
            }
            prevColumnKeyRef.current = selectedColumns.map((column) => column.field).join('|');
            initialLoadDone.current = true;
        } catch (err) {
            setError(err.message || 'Failed to load trend analytics');
        } finally {
            setIsLoading(false);
        }
    }, [applyFilterOptions, applyTrendResponse, buildInitialFiltersFromResponse, selectedColumns]);

    useEffect(() => {
        loadInitial();
        loadSavedFilters();
    }, [loadInitial, loadSavedFilters]);

    useEffect(() => {
        if (skipColumnEffectRef.current) return;
        if (!initialLoadDone.current) return;

        const columnsChanged = prevColumnKeyRef.current !== selectedColumnKey;
        prevColumnKeyRef.current = selectedColumnKey;
        if (!columnsChanged) return;

        const { filters: appliedFilters, selectedColumns: appliedColumns } = appliedQueryRef.current;
        fetchTrends(appliedFilters, appliedColumns);
    }, [selectedColumnKey, fetchTrends]);

    const handleFilterChange = useCallback((filterKey, values) => {
        setDraftFilters((prev) => ({ ...prev, [filterKey]: values }));
    }, []);

    const applyFilters = useCallback(async () => {
        const nextFilters = sanitizeFilters(draftFilters);
        initialLoadDone.current = true;
        setFilters(nextFilters);
        await fetchTrends(nextFilters, selectedColumns);
    }, [draftFilters, fetchTrends, selectedColumns]);

    const clearAllFilters = useCallback(() => {
        setDraftFilters({ ...EMPTY_SUPERVISORY_FILTERS });
    }, []);

    const applySavedFilter = useCallback(async (saved) => {
        if (!saved) return;
        initialLoadDone.current = true;
        const nextFilters = sanitizeFilters(saved.filters);
        setFilters(nextFilters);
        setDraftFilters(nextFilters);

        let nextSelectedColumns = selectedColumns;
        if (Array.isArray(saved.group_by) && saved.group_by.length > 0) {
            const resolvedColumns = saved.group_by
                .map((field) => AVAILABLE_COLUMNS.find((column) => column.field === field))
                .filter(Boolean);
            if (resolvedColumns.length > 0) {
                skipColumnEffectRef.current = true;
                setSelectedColumns(resolvedColumns);
                nextSelectedColumns = resolvedColumns;
            }
        }

        await fetchTrends(nextFilters, nextSelectedColumns);
        prevColumnKeyRef.current = nextSelectedColumns.map((column) => column.field).join('|');
        skipColumnEffectRef.current = false;
    }, [fetchTrends, selectedColumns]);

    const handleSavedFilterSelect = useCallback((event) => {
        const value = event.target.value;
        setSelectedSavedFilterId(value);
        const selected = savedFilters.find((item) => item.id === value);
        applySavedFilter(selected);
    }, [applySavedFilter, savedFilters]);

    const handleSaveCurrentFilter = useCallback(async () => {
        if (!savedFilterName.trim()) return;
        setIsSavingFilter(true);
        try {
            const response = await ApiService.saveSupervisoryFilter({
                name: savedFilterName.trim(),
                filters: sanitizeFilters(draftFilters),
                group_by: selectedColumns.map((column) => column.field),
                created_by: currentUser?.id || currentUser?.username || 'anonymous',
                created_by_name: currentUser?.name || currentUser?.username || 'anonymous'
            });
            if (response.success) {
                setSavedFilters((prev) => [response.filter, ...prev]);
                setSavedFilterName('');
                setSelectedSavedFilterId(response.filter.id);
            }
        } catch (err) {
            console.error('Failed to save trend filter', err);
        } finally {
            setIsSavingFilter(false);
        }
    }, [currentUser, draftFilters, savedFilterName, selectedColumns]);

    const forceRefresh = useCallback(async () => {
        await loadInitial();
    }, [loadInitial]);

    return {
        filterOptions,
        filters,
        draftFilters,
        savedFilters,
        selectedSavedFilterId,
        savedFilterName,
        setSavedFilterName,
        isSavingFilter,
        selectedColumns,
        setSelectedColumns,
        trendData,
        totalRecords,
        filteredRecords,
        isLoading,
        error,
        hasActiveFilters,
        hasPendingFilterChanges,
        forceRefresh,
        handleFilterChange,
        applyFilters,
        clearAllFilters,
        handleSavedFilterSelect,
        handleSaveCurrentFilter
    };
};
