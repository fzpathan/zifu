class Message(BaseModel):
    role: str
    content: str


class ChatRequest(BaseModel):
    model: str = Field(default="minimax/minimax-m2.5:free", description="Model name to use")
    temperature: float = Field(default=0.7, ge=0.0, le=2.0, description="Temperature for response randomness")
    messages: List[Message] = Field(..., description="Conversation messages")
    max_tokens: Optional[int] = Field(default=4096, description="Maximum tokens in response")
    environment: str = Field(default="dev", description="Environment (dev, poc, prod)")
    active_skills: List[str] = Field(default_factory=list, description="Optional assistant skills to activate")
    user_id: Optional[str] = Field(default=None, description="User ID — scopes all file tool calls to the user's workspace")


def detect_requested_skills(request: ChatRequest, user_message: str) -> List[str]:
    requested = {skill.lower() for skill in request.active_skills}
    normalized_message = user_message.lower()

    for skill_name, skill_config in SKILL_ALIASES.items():
        if skill_name in requested:
            requested.add(skill_name)
            continue
        if any(trigger in normalized_message for trigger in skill_config["triggers"]):
            requested.add(skill_name)

    return [skill for skill in requested if skill in SKILL_ALIASES]

def build_default_skill_system_prompt(active_skills: Optional[List[str]] = None) -> str:
    """Build a compact system prompt covering only the active skills."""
    if active_skills:
        skill_list = ", ".join(active_skills)
        contracts = [
            _SKILL_OUTPUT_CONTRACTS[s]
            for s in active_skills
            if s in _SKILL_OUTPUT_CONTRACTS
        ]
        # deduplicate (html_artifact / dashboard_artifact share the same contract)
        seen: set = set()
        unique_contracts = [c for c in contracts if not (c in seen or seen.add(c))]  # type: ignore[func-returns-value]
        contracts_block = ("\n\nOutput contracts:\n" + "\n".join(unique_contracts)) if unique_contracts else ""
        return (
            f"Active skills: {skill_list}. Use them when the request benefits from that format."
            f"{contracts_block}"
            "\n\nUse only data present or safely inferable. If data is missing, ask instead of inventing."
        )
    # No skills active — minimal prompt, no skill listing or contracts
    return "Answer clearly and concisely. Use markdown tables where tabular data helps; do not wrap tables in code blocks."


def _parse_json_object(content: str):
    if not content:
        return None

    cleaned = content.strip()
    if "```json" in cleaned:
        cleaned = cleaned.split("```json", 1)[1].split("```", 1)[0].strip()
    elif "```" in cleaned:
        cleaned = cleaned.split("```", 1)[1].split("```", 1)[0].strip()

    try:
        parsed = json.loads(cleaned)
        return parsed if isinstance(parsed, dict) else None
    except json.JSONDecodeError:
        start = cleaned.find("{")
        end = cleaned.rfind("}")
        if start != -1 and end != -1 and end > start:
            try:
                parsed = json.loads(cleaned[start:end + 1])
                return parsed if isinstance(parsed, dict) else None
            except json.JSONDecodeError:
                return None
    return None


def _render_completeness_markdown(instruction_data: Dict[str, Any]) -> str:
    run_parameters = instruction_data.get("run_parameters", [])
    node_instructions = instruction_data.get("node_instructions", [])
    missing_information = instruction_data.get("missing_information", [])

    lines = ["## Run Parameters", ""]
    if run_parameters:
        lines.extend(["| Parameter | Required | Instruction |", "|---|---:|---|"])
        for param in run_parameters:
            lines.append(f"| {param.get('name', '')} | {param.get('required', True)} | {param.get('instruction', '')} |")
    else:
        lines.append("No run parameters were identified.")

    lines.extend(["", "## Node Instructions", ""])
    if node_instructions:
        lines.extend([
            "| Order | Node ID | Node | Inputs Needed | Instruction | Validation | Expected Output |",
            "|---:|---|---|---|---|---|---|"
        ])
        for node in node_instructions:
            inputs = ", ".join(node.get("inputs_needed", []))
            validation = "<br>".join(node.get("validation", []))
            lines.append(
                f"| {node.get('order', '')} | {node.get('node_id', '')} | {node.get('node_name', '')} | "
                f"{inputs} | {node.get('instruction', '')} | {validation} | {node.get('expected_output', '')} |"
            )
    else:
        lines.append("No node instructions were generated.")

    if missing_information:
        lines.extend(["", "## Missing Information", ""])
        for item in missing_information:
            lines.append(f"- {item}")

    return "\n".join(lines)


def _validate_skill_plan(skill_name: str, plan_data: Any) -> bool:
    """Validate that skill plan contains expected keys based on skill type."""
    if not isinstance(plan_data, dict):
        return False

    if SKILL_ALIASES.get(skill_name, {}).get("skill_format") == "markdown":
        return all(key in plan_data for key in ["should_use_markdown_skill", "skill_name"])

    required_keys = {
        "field_flow": ["should_use_field_flow"],
        "flowchart": ["should_flowchart", "title"],
        "tools": ["should_use_tools"],
        "html_artifact": ["should_create_html_artifact", "title"],
        "dashboard_artifact": ["should_create_dashboard_artifact", "title"],
        "form_artifact": ["should_create_form_artifact", "title"],
        "completeness_steps": ["goal", "required_parameters", "selected_nodes", "missing_information"],
    }

    required = required_keys.get(skill_name, [])
    return all(key in plan_data for key in required)


def _fallback_skill_plan(skill_name: str, user_message: str) -> Optional[Dict[str, Any]]:
    if skill_name == "html_artifact":
        return {
            "should_create_html_artifact": True,
            "artifact_type": "mini_app",
            "title": "HTML Artifact",
            "interaction_level": "interactive",
            "layout_model": "other",
            "data_needed": [],
            "visual_direction": "Professional, self-contained browser-native HTML artifact.",
            "acceptance_checks": ["Render in the assistant preview", "Use inline CSS and JavaScript only"],
            "fallback_plan": True,
        }

    if skill_name == "dashboard_artifact":
        return {
            "should_create_dashboard_artifact": True,
            "title": "Dashboard Artifact",
            "metrics": [],
            "sections": ["summary", "exceptions", "details"],
            "data_needed": [],
            "visual_direction": "Dense, readable internal operations dashboard.",
            "fallback_plan": True,
        }

    if skill_name == "form_artifact":
        return {
            "should_create_form_artifact": True,
            "title": "Interactive Form Artifact",
            "inputs": [],
            "outputs": [],
            "data_needed": [],
            "interaction_notes": "Client-side validation, calculated output, and reset behavior.",
            "fallback_plan": True,
        }

    if SKILL_ALIASES.get(skill_name, {}).get("skill_format") == "markdown":
        return {
            "should_use_markdown_skill": True,
            "skill_name": skill_name,
            "reason": "The user explicitly requested or triggered this skill.",
            "data_needed": [],
            "output_mode": "answer",
            "fallback_plan": True,
        }

    return None


async def _plan_single_skill(
    client: AsyncOpenAI,
    request: ChatRequest,
    skill_name: str,
    user_message: str,
) -> tuple[str, str]:
    """Plan one skill and return (skill_name, plan_json). Runs independently so callers can gather."""
    builder = get_skill_builder(skill_name)
    planning_fn = builder.get("planning")

    if planning_fn is None:
        return skill_name, "{}"

    try:
        if skill_name == "completeness_steps":
            plan_result = await client.chat.completions.create(
                model=request.model,
                messages=[{"role": "user", "content": planning_fn(user_message)}],
                temperature=0.2,
                max_tokens=900,
            )
            plan_content = extract_message_content(plan_result.choices[0].message.content)
            logger.info(f"Skill '{skill_name}' raw planning response: {repr(plan_content[:500])}")
            completeness_plan = _parse_json_object(plan_content) or {
                "goal": user_message,
                "required_parameters": ["expectedRunDate", "inputConfigFilePath", "srcFilePath", "tgtFilePath"],
                "selected_nodes": [],
                "missing_information": []
            }

            if not isinstance(completeness_plan, dict) or "goal" not in completeness_plan:
                logger.warning(f"Invalid completeness_plan for {skill_name}, using default")
                completeness_plan = {
                    "goal": user_message,
                    "required_parameters": [],
                    "selected_nodes": [],
                    "missing_information": ["Failed to parse planning response"]
                }

            instruction_fn = builder.get("instruction")
            if instruction_fn:
                instruction_result = await client.chat.completions.create(
                    model=request.model,
                    messages=[{"role": "user", "content": instruction_fn(user_message, completeness_plan)}],
                    temperature=0.2,
                    max_tokens=2200,
                )
                instruction_content = extract_message_content(instruction_result.choices[0].message.content)
                instruction_data = _parse_json_object(instruction_content)

                if instruction_data:
                    return skill_name, json.dumps({
                        "plan": completeness_plan,
                        "instructions": instruction_data,
                        "markdown": _render_completeness_markdown(instruction_data)
                    }, indent=2)
                return skill_name, instruction_content
        else:
            planning_prompt = planning_fn(user_message)
            logger.info(f"Skill '{skill_name}' planning prompt: {repr(planning_prompt[:300])}")
            result = await client.chat.completions.create(
                model=request.model,
                messages=[{"role": "user", "content": planning_prompt}],
                temperature=0.2,
                max_tokens=1500,
            )
            plan_content = extract_message_content(result.choices[0].message.content)
            logger.info(f"Skill '{skill_name}' raw planning response: {repr(plan_content[:500])}")
            parsed_plan = _parse_json_object(plan_content)
            logger.info(f"Skill '{skill_name}' parsed plan: {parsed_plan}")

            if parsed_plan is None or not _validate_skill_plan(skill_name, parsed_plan):
                logger.warning(f"Invalid or incomplete planning response for {skill_name}: {plan_content[:200] if plan_content else 'empty'}")
                fallback_plan = _fallback_skill_plan(skill_name, user_message)
                return skill_name, json.dumps(fallback_plan) if fallback_plan else "{}"
            if parsed_plan.get("should_use_markdown_skill") is False:
                logger.info(f"Skill '{skill_name}' planner declined this request")
                return skill_name, "{}"
            return skill_name, json.dumps(parsed_plan)

    except Exception as e:
        logger.error(f"Failed to run planning for skill '{skill_name}': {e}", exc_info=True)
        fallback_plan = _fallback_skill_plan(skill_name, user_message)
        return skill_name, json.dumps(fallback_plan) if fallback_plan else "{}"

    return skill_name, "{}"


async def run_skill_planning_turn(client: AsyncOpenAI, request: ChatRequest, user_message: str, skills: List[str]) -> Dict[str, str]:
    """Run planning for all triggered skills concurrently."""
    skills_with_planners = [s for s in skills if get_skill_builder(s).get("planning") is not None]
    if not skills_with_planners:
        return {}

    results = await asyncio.gather(
        *[_plan_single_skill(client, request, s, user_message) for s in skills_with_planners],
        return_exceptions=True,
    )

    plans: Dict[str, str] = {}
    for item in results:
        if isinstance(item, Exception):
            logger.error(f"Skill planning task raised an exception: {item}", exc_info=item)
            continue
        skill_name, plan_content = item
        plans[skill_name] = plan_content

    return plans


def build_skill_system_messages(skill_plans: Dict[str, str]) -> List[Dict[str, str]]:
    messages = []

    for skill_name, plan_content in skill_plans.items():
        logger.info(f"Building system message for skill '{skill_name}', plan_content: {repr(plan_content[:200] if plan_content else 'empty')}")
        if not plan_content or plan_content == "{}":
            logger.info(f"Skipping skill '{skill_name}' - empty plan")
            continue

        builder = get_skill_builder(skill_name)
        system_fn = builder.get("system")

        if system_fn is None:
            logger.info(f"Skipping skill '{skill_name}' - no system_fn")
            continue

        try:
            parsed_plan = json.loads(plan_content)
            logger.info(f"Calling system_fn for skill '{skill_name}' with parsed plan: {parsed_plan}")
            system_prompt = system_fn(plan_content)
            logger.info(f"Skill '{skill_name}' system prompt (first 200 chars): {repr(system_prompt[:200])}")
            messages.append({"role": "system", "content": system_prompt})
        except json.JSONDecodeError as e:
            logger.error(f"Invalid JSON in skill plan for '{skill_name}': {e}, content: {repr(plan_content[:200])}")
        except Exception as e:
            logger.error(f"Failed to build system prompt for skill '{skill_name}': {e}")

    return messages

_openai_client: Optional[AsyncOpenAI] = None


def get_openai_client() -> AsyncOpenAI:
    """Return the module-level AsyncOpenAI singleton, creating it on first call.

    Reusing one client across requests preserves the connection pool and avoids
    the overhead of TLS handshakes on every chat request.
    """
    global _openai_client
    if _openai_client is None:
        _openai_client = AsyncOpenAI(
            api_key=get_openai_api_key(),
            base_url=get_api_base_url(),
            timeout=120,
        )
    return _openai_client


def extract_message_content(message_content: Any) -> str:
    if isinstance(message_content, str):
        return message_content

    if isinstance(message_content, list):
        text_parts = []
        for item in message_content:
            if isinstance(item, dict):
                if item.get("type") == "text" and item.get("text"):
                    text_parts.append(item["text"])
                elif item.get("type") == "output" and item.get("output"):
                    text_parts.append(item["output"])
            else:
                item_type = getattr(item, "type", None)
                item_text = getattr(item, "text", None) or getattr(item, "output", None)
                if item_type in ("text", "output") and item_text:
                    text_parts.append(item_text)
        return "".join(text_parts)

    if hasattr(message_content, "content"):
        content = message_content.content
        if isinstance(content, str):
            return content
        if isinstance(content, list):
            return extract_message_content(content)

    return ""


async def emit_sse_chunks(content: str):
    for i in range(0, len(content), SSE_CHUNK_SIZE):
        chunk_content = content[i:i + SSE_CHUNK_SIZE]
        yield f"data: {json.dumps({'content': chunk_content})}\n\n"
        await asyncio.sleep(0)


def _should_enable_tools(requested_skills: List[str], skill_plans: Dict[str, str], user_message: str) -> bool:
    if "tools" in requested_skills:
        return True

    # Some skills, like field_flow, need read_excel even when the user did not
    # explicitly say "use tools".
    if "field_flow" in requested_skills:
        return True

    if "xlsx" in requested_skills:
        return True

    tool_words = (
        "read file", "write file", "read excel", "read xlsx", "spreadsheet",
        "run command", "execute command", "shell", "grep", "glob",
        "find files", "search files", "call api", "api call",
    )
    normalized = user_message.lower()
    if any(word in normalized for word in tool_words):
        return True

    for plan_content in skill_plans.values():
        plan = _parse_json_object(plan_content)
        if not plan:
            continue
        if plan.get("should_use_tools") or plan.get("excel_file"):
            return True
        if plan.get("skill_name") == "xlsx":
            return True
        if plan.get("skill_name") == "web-artifacts-builder" and plan.get("output_mode") in ("tool_use", "file_artifact"):
            return True

    return False


def _tool_call_to_message(tool_call: Any) -> Dict[str, Any]:
    return {
        "id": tool_call.id,
        "type": "function",
        "function": {
            "name": tool_call.function.name,
            "arguments": tool_call.function.arguments or "{}",
        },
    }


def _parse_tool_arguments(raw_arguments: str) -> Dict[str, Any]:
    if not raw_arguments:
        return {}

    try:
        parsed = json.loads(raw_arguments)
        return parsed if isinstance(parsed, dict) else {}
    except json.JSONDecodeError:
        parsed = _parse_json_object(raw_arguments)
        return parsed or {}


def _tool_call_label(tool_name: str, tool_args: Dict[str, Any]) -> str:
    """Return a human-readable status label for a tool call."""
    if tool_name == "read":
        return f"Reading file: {tool_args.get('filePath', '')}"
    if tool_name == "write":
        return f"Writing file: {tool_args.get('filePath', '')}"
    if tool_name == "read_excel":
        return f"Reading Excel: {tool_args.get('filePath', '')}"
    if tool_name == "write_excel":
        return f"Writing Excel: {tool_args.get('filePath', '')}"
    if tool_name == "shell":
        cmd = tool_args.get('command', '')
        return f"Running command: {cmd[:60]}{'…' if len(cmd) > 60 else ''}"
    if tool_name == "glob":
        return f"Searching files: {tool_args.get('pattern', '')}"
    if tool_name == "grep":
        return f"Searching for: {tool_args.get('pattern', '')}"
    if tool_name == "api_call":
        return f"Calling API: {tool_args.get('method', 'GET')} {tool_args.get('endpoint', '')}"
    return f"Running tool: {tool_name}"


async def run_tool_enabled_chat(
    client: AsyncOpenAI,
    request: ChatRequest,
    provider_messages: List[Dict[str, Any]],
    workspace_path: Optional[Path] = None,
):
    """Async generator yielding tool_call status events and a final content event.

    Budget is tracked in tool calls consumed, not turns taken.  A turn that
    makes 5 parallel calls costs 5 units; a turn with 1 call costs 1 unit.
    When the budget hits zero the loop stops after completing the current
    batch — we never abandon in-flight calls — and forces a final response.
    """
    messages = list(provider_messages)
    usage_acc: Dict[str, int] = {"prompt_tokens": 0, "completion_tokens": 0, "total_tokens": 0}
    calls_remaining = MAX_TOOL_CALLS
    budget_exhausted = False
    turn = 0

    while calls_remaining > 0:
        turn += 1
        result = await client.chat.completions.create(
            model=request.model,
            messages=messages,
            temperature=request.temperature,
            max_tokens=request.max_tokens,
            tools=TOOL_DEFINITIONS,
            tool_choice="auto",
        )
        _accumulate_usage(usage_acc, result.usage)
        choice = result.choices[0]
        message = choice.message
        tool_calls = message.tool_calls or []

        logger.info(
            "Tool chat turn %d: finish_reason=%s tool_calls=%d calls_remaining=%d",
            turn, choice.finish_reason, len(tool_calls), calls_remaining,
        )

        if not tool_calls:
            # Model finished naturally — no budget prompt needed.
            yield {"type": "final", "content": extract_message_content(message.content), "usage": usage_acc}
            return

        messages.append({
            "role": "assistant",
            "content": message.content or "",
            "tool_calls": [_tool_call_to_message(tc) for tc in tool_calls],
        })

        # Parse args once; emit all labels immediately so the frontend shows
        # every pending tool before any of them starts executing.
        parsed_calls = [
            (tc, tc.function.name, _parse_tool_arguments(tc.function.arguments))
            for tc in tool_calls
        ]
        for tc, name, targs in parsed_calls:
            yield {"type": "tool_call", "tool": name, "label": _tool_call_label(name, targs)}

        # Execute all tools in this batch concurrently.
        async def _run_tool(tc, name, targs):
            logger.info("Executing tool '%s' with args: %s", name, targs)
            tool_result = await execute_tool(name, targs, workspace_path=workspace_path)
            logger.info("Tool '%s' result preview: %s", name, tool_result[:200])
            return tc.id, tool_result

        tool_results = await asyncio.gather(
            *[_run_tool(tc, name, targs) for tc, name, targs in parsed_calls]
        )

        for call_id, tool_result in tool_results:
            messages.append({
                "role": "tool",
                "tool_call_id": call_id,
                "content": tool_result,
            })

        # Deduct after completing the batch so we never abandon in-flight calls.
        calls_remaining -= len(tool_calls)
        if calls_remaining <= 0:
            budget_exhausted = True
            logger.warning(
                "Tool call budget exhausted after %d call(s) across %d turn(s) — forcing final response",
                MAX_TOOL_CALLS, turn,
            )
            break

    # Only reached when the budget ran out while the model was still calling tools.
    # budget_exhausted=False here means MAX_TOOL_CALLS=0 — skip the hint entirely.
    final_messages = list(messages)
    if budget_exhausted:
        final_messages.append({
            "role": "system",
            "content": "Tool call budget exhausted. Summarize the available tool results and clearly mention anything still missing.",
        })

    result = await client.chat.completions.create(
        model=request.model,
        messages=final_messages,
        temperature=request.temperature,
        max_tokens=request.max_tokens,
    )
    _accumulate_usage(usage_acc, result.usage)
    yield {"type": "final", "content": extract_message_content(result.choices[0].message.content), "usage": usage_acc}


async def judge_response(
    user_message: str,
    ai_response: str,
    criteria_template: str = DEFAULT_CRITERIA_TEMPLATE,
    model: str = JUDGMENT_MODEL
) -> Optional[Dict[str, Any]]:
    if not JUDGMENT_ENABLED:
        return None

    if criteria_template not in CRITERIA_TEMPLATES:
        criteria_template = DEFAULT_CRITERIA_TEMPLATE

    criteria = CRITERIA_TEMPLATES[criteria_template]
    criteria_list = "\n".join([f"- {c}" for c in criteria["criteria"]])

    judge_prompt = f"""You are an expert evaluator. Judge the following AI response based on these criteria:

{criteria_list}

User's Question:
{user_message}

AI Response to Judge:
{ai_response}

Provide your evaluation in JSON format:
{{
  "score": <0-100>,
  "reasoning": "<brief explanation>",
  "passed": <true if score >= 60, false otherwise>
}}

Be objective and fair. Consider the question's complexity when evaluating."""

    try:
        start_time = asyncio.get_running_loop().time()
        client = get_openai_client()
        result = await client.chat.completions.create(
            model=model,
            messages=[{"role": "user", "content": judge_prompt}],
            temperature=0.3,
            max_tokens=500,
        )
        content = extract_message_content(result.choices[0].message.content)

        if "```json" in content:
            content = content.split("```json")[1].split("```")[0].strip()
        elif "```" in content:
            content = content.split("```")[1].split("```")[0].strip()

        judgment = json.loads(content.strip())

        duration_ms = int((asyncio.get_running_loop().time() - start_time) * 1000)
        judgment["judge_model"] = model
        judgment["judgment_duration_ms"] = duration_ms

        logger.info(f"Judged response: score={judgment.get('score')}, passed={judgment.get('passed')}")
        return judgment

    except json.JSONDecodeError as e:
        logger.error(f"Failed to parse judgment JSON: {str(e)}")
        return None
    except Exception as e:
        logger.error(f"Judgment error: {str(e)}")
        return None


def _estimate_tokens(messages: List[Message]) -> int:
    """Rough token estimate: 1 token ≈ 4 characters (good enough for threshold checks)."""
    return sum(len(m.content) for m in messages) // 4


async def _compact_messages(
    client: AsyncOpenAI,
    model: str,
    messages: List[Message],
) -> tuple[List[Message], bool]:
    """Summarise older messages when estimated tokens exceed COMPACTION_THRESHOLD.

    Returns (messages, was_compacted).  The oldest messages are replaced with a
    single system-role summary; recent messages are kept verbatim so the model
    retains immediate context.
    """
    estimated = _estimate_tokens(messages)
    threshold = int(CONTEXT_MAX_TOKENS * COMPACTION_THRESHOLD)

    if estimated <= threshold:
        return messages, False

    # Work out how many recent messages fit inside the target budget.
    target_chars = int(CONTEXT_MAX_TOKENS * COMPACTION_TARGET) * 4
    recent: List[Message] = []
    chars_kept = 0
    for msg in reversed(messages):
        msg_len = len(msg.content)
        if chars_kept + msg_len <= target_chars:
            recent.insert(0, msg)
            chars_kept += msg_len
        else:
            break

    to_summarise = messages[: len(messages) - len(recent)]
    if not to_summarise:
        return messages, False

    history_text = "\n\n".join(
        f"{m.role.upper()}: {m.content}" for m in to_summarise
    )

    logger.info(
        "Compacting %d messages (~%d estimated tokens, threshold %d). Keeping %d recent.",
        len(to_summarise), estimated, threshold, len(recent),
    )

    try:
        summary_result = await client.chat.completions.create(
            model=model,
            messages=[{
                "role": "user",
                "content": (
                    "Summarise the following conversation history. "
                    "Preserve key facts, file paths, errors, decisions, and any "
                    "context needed to continue the conversation. Be concise.\n\n"
                    + history_text
                ),
            }],
            temperature=0.1,
            max_tokens=600,
        )
        summary_text = extract_message_content(summary_result.choices[0].message.content)
    except Exception as e:
        logger.error("Compaction summary call failed: %s — skipping compaction", e)
        return messages, False

    summary_msg = Message(
        role="system",
        content=f"[Earlier conversation compacted — summary]\n{summary_text}",
    )

    logger.info("Compaction complete. %d → 1 summary + %d recent messages.", len(to_summarise), len(recent))
    return [summary_msg] + recent, True


def _accumulate_usage(acc: Dict[str, int], usage) -> None:
    """Add provider usage counters into a running total dict."""
    if not usage:
        return
    acc["prompt_tokens"]     += getattr(usage, "prompt_tokens",     0) or 0
    acc["completion_tokens"] += getattr(usage, "completion_tokens", 0) or 0
    acc["total_tokens"]      += getattr(usage, "total_tokens",      0) or 0


def _trim_messages(messages: List[Message]) -> List[Message]:
    """Keep only the most recent MAX_HISTORY_MESSAGES to avoid exceeding token limits."""
    if len(messages) <= MAX_HISTORY_MESSAGES:
        return messages
    trimmed = len(messages) - MAX_HISTORY_MESSAGES
    logger.info("Trimming %d old message(s) from history (keeping last %d)", trimmed, MAX_HISTORY_MESSAGES)
    return messages[-MAX_HISTORY_MESSAGES:]


async def stream_chat_response(request: ChatRequest):
    full_response = ""
    user_message = ""

    try:
        client = get_openai_client()

        for msg in reversed(request.messages):
            if msg.role == "user":
                user_message = msg.content
                break

        requested_skills = detect_requested_skills(request, user_message)

        logger.info(
            "AI Chat request: model=%s temp=%s messages=%d env=%s skills=%s",
            request.model, request.temperature, len(request.messages),
            request.environment, requested_skills,
        )

        skill_plans = await run_skill_planning_turn(client, request, user_message, requested_skills) if requested_skills else {}
        skill_system_messages = build_skill_system_messages(skill_plans)

        compacted_messages, was_compacted = await _compact_messages(client, request.model, request.messages)
        trimmed_messages = _trim_messages(compacted_messages)

        workspace_path = ensure_user_workspace(request.user_id) if request.user_id else None
        system_content = build_default_skill_system_prompt(requested_skills or None)
        if workspace_path:
            system_content += f"\n\nWorkspace: {workspace_path}. Use relative file paths for all file operations."

        rag_kb = get_rag_kb()
        rag_chunks = await rag_kb.async_search(user_message, top_k=3)
        if rag_chunks:
            rag_context = rag_kb.format_context(rag_chunks)
            system_content = rag_context + "\n\n" + system_content
            logger.info("RAG: injected %d chunk(s) for query: %s", len(rag_chunks), user_message[:80])

        provider_messages = [
            {"role": "system", "content": system_content},
            *skill_system_messages,
            *[{"role": m.role, "content": m.content} for m in trimmed_messages]
        ]

        usage: Dict[str, int] = {"prompt_tokens": 0, "completion_tokens": 0, "total_tokens": 0}

        if _should_enable_tools(requested_skills, skill_plans, user_message):
            async for event in run_tool_enabled_chat(client, request, provider_messages, workspace_path=workspace_path):
                if event["type"] == "tool_call":
                    yield f"data: {json.dumps({'tool_call': event['tool'], 'tool_label': event['label']})}\n\n"
                elif event["type"] == "final":
                    full_response = event["content"]
                    event_usage = event.get("usage") or {}
                    usage["prompt_tokens"]     = event_usage.get("prompt_tokens", 0)
                    usage["completion_tokens"] = event_usage.get("completion_tokens", 0)
                    usage["total_tokens"]      = event_usage.get("total_tokens", 0)
        else:
            result = await client.chat.completions.create(
                model=request.model,
                messages=provider_messages,
                temperature=request.temperature,
                max_tokens=request.max_tokens,
            )
            full_response = extract_message_content(result.choices[0].message.content)
            _accumulate_usage(usage, result.usage)

        if not full_response:
            logger.warning("AI provider returned an empty response")
            yield f"data: {json.dumps({'error': 'Empty response from AI provider'})}\n\n"
            return

        async for chunk in emit_sse_chunks(full_response):
            yield chunk

        yield f"data: {json.dumps({'done': True, 'usage': usage, 'compacted': was_compacted})}\n\n"

        # Sample judgments — running on every response doubles API cost for no extra signal.
        if JUDGMENT_ENABLED and full_response and user_message and random.random() < JUDGMENT_SAMPLE_RATE:
            asyncio.create_task(
                _silent_judge_and_save(
                    user_message=user_message,
                    ai_response=full_response,
                    model=request.model,
                    temperature=request.temperature,
                    environment=request.environment,
                )
            )

    except HTTPException as e:
        yield f"data: {json.dumps({'error': e.detail})}\n\n"
    except Exception as e:
        logger.error(f"AI Chat error: {str(e)}")
        yield f"data: {json.dumps({'error': str(e)})}\n\n"


async def _silent_judge_and_save(
    user_message: str,
    ai_response: str,
    model: str,
    temperature: float,
    environment: str = "dev",
):
    try:
        from .judgment_service import JudgmentService
        import time

        judgment = await judge_response(
            user_message=user_message,
            ai_response=ai_response,
            criteria_template=DEFAULT_CRITERIA_TEMPLATE,
            model=JUDGMENT_MODEL
        )

        if judgment:
            judgment_data = {
                "judgment_id": f"judgment-{int(time.time())}-{os.urandom(4).hex()}",
                "chat_id": "unknown",
                "message_id": "unknown",
                "timestamp": datetime.now().isoformat() + "Z",
                "model_used": model,
                "criteria_template": DEFAULT_CRITERIA_TEMPLATE,
                "criteria_list": CRITERIA_TEMPLATES[DEFAULT_CRITERIA_TEMPLATE]["criteria"],
                "user_message": user_message,
                "ai_response": ai_response,
                "judgment": judgment,
                "metadata": {
                    "temperature": temperature,
                    "response_length": len(ai_response),
                    "user_id": "anonymous",
                    "environment": environment,
                }
            }
            JudgmentService.save_judgment(judgment_data)
            logger.info(f"Silently judged and saved: score={judgment.get('score')}")

    except Exception as e:
        logger.error(f"Silent judgment failed: {str(e)}")

Search pipeline
---------------
1. Keyword pre-filter  (BM25-lite, always runs, zero cost)
   → top RAG_KEYWORD_PRE_K candidates
2. Embedding re-rank   (optional, configured via env var)
   → cosine similarity, returns top_k final results

Embedding providers (set RAG_EMBEDDING_PROVIDER):
  "keyword_only"         – default, no embeddings (steps 1 only)
  "openrouter"           – OpenAI-compatible embeddings API (OpenRouter)
  "sentence_transformer" – local model, zero API cost, runs on CPU

OpenRouter / OpenAI provider env vars:
  OPENAI_API_KEY         (shared with ai_assistant.py)
  OPENAI_API_BASE_URL    (shared with ai_assistant.py)

Sentence-transformer env vars:
  RAG_ST_MODEL           default "all-MiniLM-L6-v2"  (~90 MB, fast on CPU)

Chunk embeddings are expensive to build (API calls or local inference).
They are cached to disk keyed by:
    SHA-256(source JSON bytes) + provider + model name

Cache is invalidated automatically when rag_assist_system_prompt.json

# ── Cosine similarity (numpy optional) ───────────────────────────────────────
try:
    import numpy as np

    def _cosine_sim(a: List[float], b: List[float]) -> float:
        va, vb = np.array(a, dtype=np.float32), np.array(b, dtype=np.float32)
        denom = np.linalg.norm(va) * np.linalg.norm(vb)
        return float(np.dot(va, vb) / denom) if denom > 1e-10 else 0.0

except ImportError:
    def _cosine_sim(a: List[float], b: List[float]) -> float:
        dot  = sum(x * y for x, y in zip(a, b))
        mag  = (sum(x * x for x in a) ** 0.5) * (sum(x * x for x in b) ** 0.5)
        return dot / mag if mag > 1e-10 else 0.0

class OpenRouterEmbeddingProvider:
    """
    OpenAI-compatible embeddings via OpenRouter (or any OpenAI endpoint).

    Shares the same OPENAI_API_KEY / OPENAI_API_BASE_URL that ai_assistant.py
    already uses, so no extra credentials are needed.

    embed_batch() uses the *sync* OpenAI client so it can be called from a
    thread-pool executor during cache-build without blocking the event loop.

    embed_query_async() uses the *async* client for per-request calls.
    """

    def __init__(self) -> None:
        self.model    = OPENROUTER_EMBED_MODEL
        self.api_key  = os.environ.get(
            "OPENAI_API_KEY",
            "",
        )
        self.base_url = os.environ.get(
            "OPENAI_API_BASE_URL", "https://openrouter.ai/api/v1"
        ).rstrip("/")
        self._async_client = None

    # ── sync (used in executor for cache build) ───────────────────────────────
    def embed_batch(self, texts: List[str]) -> List[List[float]]:
        from openai import OpenAI
        client   = OpenAI(api_key=self.api_key, base_url=self.base_url, timeout=60)
        response = client.embeddings.create(model=self.model, input=texts)
        return [item.embedding for item in sorted(response.data, key=lambda x: x.index)]

    # ── async (used at query time) ────────────────────────────────────────────
    async def embed_query_async(self, query: str) -> List[float]:
        from openai import AsyncOpenAI
        if self._async_client is None:
            self._async_client = AsyncOpenAI(
                api_key=self.api_key, base_url=self.base_url, timeout=30
            )
        response = await self._async_client.embeddings.create(
            model=self.model, input=[query]
        )
        return response.data[0].embedding


class SentenceTransformerProvider:
    """
    Local embedding model via sentence-transformers.

    The model is downloaded once to the HuggingFace cache (~90 MB for
    all-MiniLM-L6-v2) and reused across restarts.  No API key required.
    CPU inference is fast enough for <50 chunks.

    Install:  pip install sentence-transformers
    """

    def __init__(self) -> None:
        self.model_name = ST_MODEL_NAME
        self._model     = None   # lazy-load on first use

    def _get_model(self):
        if self._model is None:
            from sentence_transformers import SentenceTransformer
            logger.info("Loading SentenceTransformer model '%s' …", self.model_name)
            self._model = SentenceTransformer(self.model_name)
            logger.info("SentenceTransformer model loaded.")
        return self._model

    # ── sync (used in executor for cache build) ───────────────────────────────
    def embed_batch(self, texts: List[str]) -> List[List[float]]:
        model = self._get_model()
        vecs  = model.encode(texts, batch_size=32, show_progress_bar=False)
        return [v.tolist() for v in vecs]

    # ── async (used at query time, runs model in thread pool) ─────────────────
    async def embed_query_async(self, query: str) -> List[float]:
        model  = self._get_model()
        result = await asyncio.to_thread(
            model.encode, [query], batch_size=1, show_progress_bar=False
        )
        return result[0].tolist()

def _source_hash() -> str:
    """SHA-256 of the RAG JSON source file.  Changes → cache miss."""
    if not RAG_JSON_PATH.exists():
        return "missing"
    return hashlib.sha256(RAG_JSON_PATH.read_bytes()).hexdigest()


def _cache_path(source_hash: str) -> Path:
    key = hashlib.sha256(
        f"{source_hash}:{EMBEDDING_PROVIDER}:{OPENROUTER_EMBED_MODEL if EMBEDDING_PROVIDER == 'openrouter' else ST_MODEL_NAME}".encode()
    ).hexdigest()[:20]
    return RAG_CACHE_DIR / f"embeddings_{key}.json"


def _load_cache(source_hash: str) -> Optional[Dict[str, List[float]]]:
    path = _cache_path(source_hash)
    if not path.exists():
        logger.info("RAG cache miss (no file at %s)", path)
        return None
    try:
        data = json.loads(path.read_text(encoding="utf-8"))
        if data.get("source_hash") != source_hash:
            logger.info("RAG cache miss (source hash mismatch)")
            return None
        logger.info("RAG cache hit: loaded %d embeddings from %s", len(data["embeddings"]), path.name)
        return data["embeddings"]   # {chunk_id: [floats]}
    except Exception as exc:
        logger.warning("RAG cache read error (%s) — will rebuild", exc)
        return None


def _save_cache(source_hash: str, embeddings: Dict[str, List[float]]) -> None:
    try:
        RAG_CACHE_DIR.mkdir(parents=True, exist_ok=True)
        path = _cache_path(source_hash)
        payload = {
            "source_hash": source_hash,
            "provider":    EMBEDDING_PROVIDER,
            "model":       OPENROUTER_EMBED_MODEL if EMBEDDING_PROVIDER == "openrouter" else ST_MODEL_NAME,
            "chunk_count": len(embeddings),
            "embeddings":  embeddings,
        }
        path.write_text(json.dumps(payload, separators=(",", ":")), encoding="utf-8")
        logger.info("RAG cache saved: %d embeddings → %s", len(embeddings), path.name)
        # clean up stale cache files (keep only the newest)
        for old in RAG_CACHE_DIR.glob("embeddings_*.json"):
            if old != path:
                try:
                    old.unlink()
                    logger.debug("Removed stale cache file %s", old.name)
                except OSError:
                    pass
    except Exception as exc:
        logger.warning("RAG cache write error (%s) — search will still work", exc)

# Generic JSON chunker

@dataclass
class RagChunk:
    id: str
    title: str            # human-readable breadcrumb
    breadcrumb: List[str] # raw path segments from root
    content: str          # "[Path: ...]\n\n<json>" — ready to inject
    keywords: List[str] = field(default_factory=list)


def _key_to_label(key: str) -> str:
    s = re.sub(r'([A-Z])', r' \1', key)
    return s.replace("_", " ").strip().title()


def _breadcrumb_str(path: List[str]) -> str:
    return " > ".join(_key_to_label(p) for p in path)


def _chunk_id(path: List[str]) -> str:
    return "__".join(re.sub(r'[^a-z0-9]', '_', p.lower()) for p in path) or "root"


def _extract_tokens(obj: Any, depth: int = 3) -> List[str]:
    tokens: List[str] = []
    if depth <= 0:
        return tokens
    if isinstance(obj, str):
        tokens += re.findall(r'[a-z][a-z0-9_]{2,}', obj.lower())
    elif isinstance(obj, dict):
        for k, v in obj.items():
            tokens += re.findall(r'[a-z][a-z0-9_]{2,}', k.lower())
            tokens += _extract_tokens(v, depth - 1)
    elif isinstance(obj, list):
        for item in obj[:20]:
            tokens += _extract_tokens(item, depth - 1)
    return tokens


def _make_keywords(path: List[str], obj: Any) -> List[str]:
    all_tokens = [seg.lower() for seg in path] + _extract_tokens(obj)
    filtered   = [t for t in all_tokens if t not in _STOP_WORDS and len(t) > 2]
    seen: set  = set()
    return [t for t in filtered if not (t in seen or seen.add(t))]  # type: ignore[func-returns-value]


def _format_content(path: List[str], obj: Any) -> str:
    breadcrumb = _breadcrumb_str(path) if path else "Root"
    return f"[Path: {breadcrumb}]\n\n{json.dumps(obj, indent=2)}"


def _is_chunkable(obj: Any, serialised: str) -> bool:
    return isinstance(obj, dict) and (
        len(obj) >= MIN_KEYS or len(serialised) >= MIN_CHUNK_CHARS
    )


def _chunk_node(obj: Any, path: List[str], out: List[RagChunk]) -> None:
    """
    Emit OR split — never both.
      size > MAX  →  split into per-child chunks (recurse into children)
      size ok     →  emit as one chunk (children captured inside the JSON)
      too small   →  skip (absorbed into nearest parent chunk)
    """
    if not isinstance(obj, dict):
        return
    serialised = json.dumps(obj, indent=2)
    if len(serialised) > MAX_CHUNK_CHARS:
        for key, child in obj.items():
            _chunk_node(child, path + [key], out)
        return
    if _is_chunkable(obj, serialised):
        out.append(RagChunk(
            id=_chunk_id(path),
            title=_breadcrumb_str(path) if path else "Document Root",
            breadcrumb=list(path),
            content=_format_content(path, obj),
            keywords=_make_keywords(path, obj),
        ))

class RagKnowledgeBase:
    def __init__(self) -> None:
        self.chunks: List[RagChunk] = []
        # embeddings dict: chunk_id → float vector (None = not yet built)
        self._embeddings: Optional[Dict[str, List[float]]] = None
        self._embedding_lock = asyncio.Lock()
        self._provider: Optional[Any] = None
        self._source_hash: str = _source_hash()
        self._load_chunks()
        # Attempt a fast cache load at startup (sync, no API call)
        if EMBEDDING_PROVIDER != "keyword_only":
            cached = _load_cache(self._source_hash)
            if cached is not None:
                self._embeddings = cached

    # ── Chunking ──────────────────────────────────────────────────────────────

    def _load_chunks(self) -> None:
        if not RAG_JSON_PATH.exists():
            logger.warning("RAG JSON not found at %s", RAG_JSON_PATH)
            return
        try:
            with open(RAG_JSON_PATH, encoding="utf-8") as fh:
                data = json.load(fh)
            _chunk_node(data, [], self.chunks)
            logger.info("RAG: %d chunks loaded from %s", len(self.chunks), RAG_JSON_PATH.name)
        except Exception as exc:
            logger.error("RAG chunk load failed: %s", exc, exc_info=True)

    # ── Provider factory ──────────────────────────────────────────────────────

    def _get_provider(self):
        if self._provider is None:
            if EMBEDDING_PROVIDER == "openrouter":
                self._provider = OpenRouterEmbeddingProvider()
                logger.info("RAG embedding provider: OpenRouter (%s)", OPENROUTER_EMBED_MODEL)
            elif EMBEDDING_PROVIDER == "sentence_transformer":
                self._provider = SentenceTransformerProvider()
                logger.info("RAG embedding provider: SentenceTransformer (%s)", ST_MODEL_NAME)
        return self._provider

    # ── Embedding build (sync, runs in executor) ──────────────────────────────

    def _build_embeddings_sync(self) -> Dict[str, List[float]]:
        """Build and cache chunk embeddings.  Always runs in a thread pool."""
        cached = _load_cache(self._source_hash)
        if cached is not None:
            return cached

        provider = self._get_provider()
        if provider is None:
            return {}

        logger.info("RAG: building embeddings for %d chunks via %s …", len(self.chunks), EMBEDDING_PROVIDER)
        texts = [c.content for c in self.chunks]
        ids   = [c.id      for c in self.chunks]
        vecs  = provider.embed_batch(texts)
        result = dict(zip(ids, vecs))
        _save_cache(self._source_hash, result)
        return result

    async def _ensure_embeddings(self) -> None:
        """Guarantee embeddings are available; builds them if not cached."""
        if self._embeddings is not None:
            return
        async with self._embedding_lock:
            if self._embeddings is not None:   # double-check after acquiring lock
                return
            try:
                self._embeddings = await asyncio.to_thread(self._build_embeddings_sync)
            except Exception as exc:
                logger.error("RAG embedding build failed: %s — falling back to keyword-only", exc, exc_info=True)
                self._embeddings = {}   # empty → no retry; keyword fallback used

    # ── Keyword search (sync, always available) ───────────────────────────────

    def search(self, query: str, top_k: int = 3) -> List[RagChunk]:
        """Pure keyword scoring — no embeddings, instant."""
        if not self.chunks or not query:
            return []
        query_terms = set(re.findall(r'[a-z][a-z0-9_]{2,}', query.lower())) - _STOP_WORDS
        if not query_terms:
            return []

        scored: List[Tuple[float, RagChunk]] = []
        for chunk in self.chunks:
            kw_overlap   = len(query_terms & set(chunk.keywords))
            title_hits   = sum(1 for t in query_terms if t in chunk.title.lower())
            bc_overlap   = len(query_terms & {seg.lower() for seg in chunk.breadcrumb})
            content_hits = min(sum(1 for t in query_terms if t in chunk.content.lower()), 10)
            score = bc_overlap * 6 + title_hits * 5 + kw_overlap * 3 + content_hits
            if score > 0:
                scored.append((score, chunk))

        scored.sort(key=lambda x: -x[0])
        return [c for _, c in scored[:top_k]]

    # ── Hybrid search (async, used by ai_assistant.py) ────────────────────────

    async def async_search(self, query: str, top_k: int = 3) -> List[RagChunk]:
        """
        Hybrid search:
          1. Keyword pre-filter → KEYWORD_PRE_K candidates
          2. Embedding re-rank  → top_k final results

        Falls back to keyword-only if embeddings are unavailable.
        """
        candidates = self.search(query, top_k=KEYWORD_PRE_K)

        if EMBEDDING_PROVIDER == "keyword_only" or not candidates:
            return candidates[:top_k]

        await self._ensure_embeddings()

        if not self._embeddings:
            logger.debug("RAG: no embeddings available — using keyword results")
            return candidates[:top_k]

        provider = self._get_provider()
        if provider is None:
            logger.warning("RAG: no embedding provider configured for '%s' — using keyword results", EMBEDDING_PROVIDER)
            return candidates[:top_k]

        try:
            query_emb = await provider.embed_query_async(query)
        except Exception as exc:
            logger.warning("RAG query embedding failed (%s) — using keyword results", exc)
            return candidates[:top_k]

        scored: List[Tuple[float, RagChunk]] = []
        for chunk in candidates:
            chunk_emb = self._embeddings.get(chunk.id)
            if chunk_emb is None:
                continue
            sim = _cosine_sim(query_emb, chunk_emb)
            scored.append((sim, chunk))

        scored.sort(key=lambda x: -x[0])
        logger.debug(
            "RAG hybrid: top scores %s",
            [(round(s, 3), c.id) for s, c in scored[:top_k]],
        )
        return [c for _, c in scored[:top_k]]

    # ── Context formatter ─────────────────────────────────────────────────────

    def format_context(self, chunks: List[RagChunk]) -> str:
        if not chunks:
            return ""
        parts = ["[ControlDash Knowledge Base - Retrieved Context]"]
        for chunk in chunks:
            parts.append(f"\n### {chunk.title}\n{chunk.content}")
        return "\n\n".join(parts)

    # ── Debug helper ──────────────────────────────────────────────────────────

    def debug_chunks(self) -> List[Dict[str, Any]]:
        return [
            {
                "id":             c.id,
                "title":          c.title,
                "breadcrumb":     c.breadcrumb,
                "keywords_count": len(c.keywords),
                "content_chars":  len(c.content),
                "has_embedding":  c.id in (self._embeddings or {}),
            }
            for c in self.chunks
        ]


# ── Singleton ─────────────────────────────────────────────────────────────────

_rag_kb: Optional[RagKnowledgeBase] = None


def get_rag_kb() -> RagKnowledgeBase:
    global _rag_kb
    if _rag_kb is None:
        _rag_kb = RagKnowledgeBase()
    return _rag_kb
