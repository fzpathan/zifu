def _accumulate_usage(acc: Dict[str, int], usage) -> None:
    if not usage:
        return
    acc["prompt_tokens"]     += getattr(usage, "prompt_tokens",     0) or 0
    acc["completion_tokens"] += getattr(usage, "completion_tokens", 0) or 0
    acc["total_tokens"]      += getattr(usage, "total_tokens",      0) or 0


def _extract_json(text: str) -> str:
    """Find and return the first complete JSON object or array from raw LLM text.

    Strips code fences first, then walks character-by-character to find the
    matching closing bracket — handles preamble text before the JSON object.
    """
    text = re.sub(r"```[a-z]*\n?|```", "", text).strip()
    for opener, closer in [('{', '}'), ('[', ']')]:
        start = text.find(opener)
        if start == -1:
            continue
        depth = 0
        in_str = False
        esc = False
        for i, ch in enumerate(text[start:], start):
            if esc:
                esc = False
                continue
            if ch == '\\' and in_str:
                esc = True
                continue
            if ch == '"':
                in_str = not in_str
            if not in_str:
                if ch == opener:
                    depth += 1
                elif ch == closer:
                    depth -= 1
                    if depth == 0:
                        return text[start:i + 1]
    return text

_CLASSIFICATION_PROMPT = """You are a workflow router. Given the user message and the list of available workflows, identify the single best matching workflow.

Available workflows:
{workflows_summary}

User message: "{message}"

Respond with JSON only (no markdown fences):
{{"workflow_id": "<id_or_null>", "confidence": 0.0}}

Rules:
- workflow_id must exactly match one of the available ids, or be null if nothing matches
- confidence: 0.0-1.0. Use null if confidence < 0.35."""


async def classify_intent(client, model: str, message: str, workflows: list) -> dict:
    parts = []
    for w in workflows:
        ic = w.get("intent_classification", {})
        examples = ic.get("examples", [])[:5]
        keywords = ic.get("keywords", [])[:10]
        parts.append(
            f"- id: {w['id']}\n  name: {w['name']}\n"
            f"  description: {w['description']}\n"
            f"  examples: {examples}\n"
            f"  keywords: {keywords}"
        )
    prompt = _CLASSIFICATION_PROMPT.format(
        workflows_summary="\n".join(parts),
        message=message,
    )
    result = await client.chat.completions.create(
        model=model,
        messages=[{"role": "user", "content": prompt}],
        temperature=0.1,
        max_tokens=120,
    )
    raw = extract_message_content(result.choices[0].message.content)
    try:
        return json.loads(_extract_json(raw)), result.usage
    except Exception:
        return {"workflow_id": None, "confidence": 0.0}, result.usage

_EXTRACTION_PROMPT = """Extract the following parameter values from the user message.

Parameters:
{params_list}

User message: "{message}"

Respond with JSON only (no markdown). Return each parameter key with its extracted string value, or null if not found.
Example: {{"file_path": "/opt/config.xlsx", "control_type": null}}"""


async def extract_params(client, model: str, message: str, params: list) -> dict:
    if not params:
        return {}
    lines = []
    for p in params:
        line = f"- key: {p['key']}, label: {p['label']}, description: {p.get('description', '')}"
        if p.get("options"):
            line += f", allowed_values: {p['options']}"
        if p.get("example"):
            line += f", example: \"{p['example']}\""
        lines.append(line)
    prompt = _EXTRACTION_PROMPT.format(
        params_list="\n".join(lines),
        message=message,
    )
    result = await client.chat.completions.create(
        model=model,
        messages=[{"role": "user", "content": prompt}],
        temperature=0.0,
        max_tokens=500,
    )
    raw = extract_message_content(result.choices[0].message.content)
    try:
        return json.loads(_extract_json(raw)), result.usage
    except Exception:
        return {p["key"]: None for p in params}, result.usage


def find_missing_params(params: list, extracted: dict) -> list:
    missing = []
    for p in params:
        if p.get("required", False):
            val = extracted.get(p["key"])
            if val is None or (isinstance(val, str) and not val.strip()):
                missing.append(p)
    return missing

async def _exec_read_file(confirmed_params: dict, prior_results: dict):
    _ensure_api_root_on_path()
    file_path = (confirmed_params.get("input_file_path") or "").strip()
    if not file_path:
        return "skipped", []

    path = Path(file_path)
    if not path.exists():
        return "error", f"File not found: {file_path}"

    try:
        import pandas as pd
        suffix = path.suffix.lower()
        if suffix in (".xlsx", ".xls"):
            sheet = confirmed_params.get("sheet_name") or 0
            df = pd.read_excel(path, sheet_name=sheet)
        elif suffix == ".csv":
            df = pd.read_csv(path)
        else:
            return "error", f"Unsupported file type '{suffix}'. Use .xlsx, .xls, or .csv"

        df = df.head(500)
        rows = df.where(df.notna(), other=None).to_dict(orient="records")
        return "success", {"rows": rows, "row_count": len(rows), "columns": list(df.columns), "file": file_path}
    except Exception as exc:
        logger.error("Read file error: %s", exc, exc_info=True)
        return "error", str(exc)


async def _exec_write_file(confirmed_params: dict, prior_results: dict):
    _ensure_api_root_on_path()
    output_path = (confirmed_params.get("output_file_path") or "").strip()
    if not output_path:
        return "skipped", "No output path specified"

    transform_result = prior_results.get("execute_transform", {}).get("value")
    if not transform_result or isinstance(transform_result, str):
        return "error", "No transform results available to write"

    try:
        import pandas as pd
        path = Path(output_path)
        path.parent.mkdir(parents=True, exist_ok=True)

        df = pd.DataFrame(transform_result if isinstance(transform_result, list) else [transform_result])
        suffix = path.suffix.lower()
        if suffix == ".csv":
            df.to_csv(path, index=False)
        else:
            if suffix not in (".xlsx", ".xls"):
                path = path.with_suffix(".xlsx")
            df.to_excel(path, index=False)

        return "success", {"written_to": str(path), "rows": len(df), "columns": list(df.columns)}
    except Exception as exc:
        logger.error("Write file error: %s", exc, exc_info=True)
        return "error", str(exc)

async def _exec_transform(confirmed_params: dict, prior_results: dict):
    _ensure_api_root_on_path()
    try:
        import pandas as pd
        from transform_operations.transform_operations import TransformOperations

        ops = TransformOperations()

        # Prefer data read from file over inline JSON
        read_result = prior_results.get("read_input_file", {})
        if read_result.get("status") == "success" and read_result.get("value"):
            file_data = read_result["value"]
            sample_data = file_data.get("rows", []) if isinstance(file_data, dict) else file_data
        else:
            sample_data = confirmed_params.get("sample_data", "[]")
            if isinstance(sample_data, str):
                sample_data = json.loads(sample_data) if sample_data.strip() else []

        if not sample_data:
            return "error", "No sample data provided. Either supply Sample Data JSON or an Input File Path."

        op_params_raw = confirmed_params.get("operation_params") or "{}"
        if isinstance(op_params_raw, str) and op_params_raw.strip():
            try:
                op_params = json.loads(op_params_raw)
            except Exception:
                op_params = {}
        elif isinstance(op_params_raw, dict):
            op_params = op_params_raw
        else:
            op_params = {}

        operation = {
            "Operation": confirmed_params.get("operation_type", ""),
            "NewFieldName": confirmed_params.get("new_field_name", ""),
            "Inputs": confirmed_params.get("input_fields", ""),
            **op_params,
        }

        df = pd.DataFrame(sample_data)
        result_df = ops.execute_operation(df, operation)
        result_df = result_df.where(result_df.notna(), other=None)
        # Convert Timestamps and other non-JSON types to strings
        for col in result_df.columns:
            if result_df[col].dtype.kind in ("M", "m"):  # datetime / timedelta
                result_df[col] = result_df[col].astype(str)
        return "success", result_df.to_dict(orient="records")
    except Exception as exc:
        logger.error("Transform step error: %s", exc, exc_info=True)
        return "error", str(exc)

async def _exec_poll_until(confirmed_params: dict, prior_results: dict, poll_config: dict):

        max_attempts = poll_config.get("max_attempts", 40)

async def _exec_config_search(confirmed_params: dict, prior_results: dict):
    _ensure_api_root_on_path()
    try:
        from routers.config_search import (
            _run_parquet_subprocess,
            get_available_types,
            get_file_path,
        )

        query = (confirmed_params.get("query") or "").strip()
        sheet_filter = (confirmed_params.get("sheet") or "").strip()
        type_filter = (confirmed_params.get("control_type") or "").strip()

        if not query:
            return "error", "Search query is required"

        available_sheets = ["InputFiles", "rules", "Enrichment", "FieldOperations", "OutputRule"]
        sheets_to_search = [sheet_filter] if sheet_filter else available_sheets
        types_to_search = [type_filter] if type_filter else get_available_types()

        results_all: List[dict] = []
        for type_name in types_to_search:
            for sheet_name in sheets_to_search:
                file_path = get_file_path(type_name, sheet_name)

                    continue

                    sr = _run_parquet_subprocess("search", {

                        "column": None,

                    for row in rows[:20]:
                        row["_type"] = type_name
                        row["_sheet"] = sheet_name
                    results_all.extend(rows[:20])
                except Exception:
                    continue

        return "success", {"results": results_all, "total": len(results_all), "query": query}
    except Exception as exc:
        logger.error("Config search step error: %s", exc, exc_info=True)
        return "error", str(exc)


async def _exec_llm_summarize(step: dict, confirmed_params: dict, prior_results: dict, client, model: str):
    """Async generator that yields (chunk_type, content) tuples."""
    try:
        prompt_template = step.get("prompt_template", "")

        def _resolve(match):
            key = match.group(1).strip()
            if "." in key:
                step_id, field = key.split(".", 1)
                step_entry = prior_results.get(step_id, {})
                val = step_entry.get("value", {})
                if isinstance(val, dict):
                    # Fall back to the full dict JSON if the specific sub-key is absent
                    resolved = val.get(field, json.dumps(val, default=str))
                elif isinstance(val, list):
                    resolved = json.dumps(val[:15], default=str)
                else:
                    resolved = str(val)
                return str(resolved)[:3000]
            else:
                val = confirmed_params.get(key, "")
                if isinstance(val, (dict, list)):
                    return json.dumps(val, default=str)[:500]
                return str(val) if val is not None else ""

        rendered = re.sub(r"\{\{([^}]+)\}\}", _resolve, prompt_template)

        # Append full prior results (with statuses) so the LLM knows what actually ran
        extra_parts = []
        for sid, sdata in prior_results.items():
            status = sdata.get("status", "unknown")
            v = sdata.get("value")
            if v is not None:
                extra_parts.append(f"[{sid}] status={status}: {json.dumps(v, default=str)[:3000]}")
        if extra_parts:
            rendered += "\n\nFull step results:\n" + "\n".join(extra_parts)

        stream = await client.chat.completions.create(
            model=model,
            messages=[{"role": "user", "content": rendered}],
            temperature=0.3,
            max_tokens=1500,
            stream=True,
            stream_options={"include_usage": True},
        )
        full = ""
        last_usage = None
        async for chunk in stream:
            if chunk.usage:
                last_usage = chunk.usage
            delta = chunk.choices[0].delta if chunk.choices else None
            if delta and delta.content:
                full += delta.content
                yield "chunk", delta.content
        yield "done", full
        yield "usage", last_usage
    except Exception as exc:
        logger.error("LLM summarize error: %s", exc, exc_info=True)
        yield "error", str(exc)

- Conditional   : Inputs=<col>, operation_params={"condition": "<expr>", "true_value": "<val>", "false_value": "<val>"}
- MapValues     : Inputs=<col>, operation_params={"mapping": {"old": "new", ...}}
- StringOperation: Inputs=<col>, operation_params={"operation": "upper|lower|strip|title|reverse"}
- NumericOperation: Inputs=<col1,col2,...>, operation_params={"operation": "abs|round|ceil|floor|sqrt|log|sum|mean|max|min", "decimal_places": 2}
- SetValue      : Inputs=<any_col>, operation_params={"value": "<fixed_value>"}
- CopyField     : Inputs=<source_col>
- Split         : Inputs=<col>, operation_params={"delimiter": "-", "index": 0}
- Replace       : Inputs=<col>, operation_params={"pattern": "old", "replacement": "new", "use_regex": false}

Respond with valid JSON only — no markdown, no text outside the JSON:
{
  "operation_type": "...",
  "new_field_name": "snake_case_output_name",
  "input_fields": "col1" or "col1,col2",
  "operation_params": { ... },
  "sample_data": [ { "col1": "val1" }, ... ],
  "explanation": "one-line plain-English description of what this formula does"
}

Rules:
- sample_data must have at least 3 rows containing ALL input field(s) with realistic values
- new_field_name must be snake_case
- operation_params must only include keys valid for the chosen operation_type
- If user supplies sample_data, use those rows exactly; otherwise generate realistic test rows"""


async def _exec_llm_build_formula(step: dict, confirmed_params: dict, prior_results: dict, client, model: str, usage_acc: dict):
    """LLM generates a full transform formula config + test data from a plain-English description."""
    description = (confirmed_params.get("formula_description") or "").strip()
    new_field_hint = (confirmed_params.get("new_field_name") or "").strip()
    sample_data_raw = confirmed_params.get("sample_data", "")

    sample_data = []
    if isinstance(sample_data_raw, str) and sample_data_raw.strip():
        try:
            sample_data = json.loads(sample_data_raw)
        except Exception:
            pass
    elif isinstance(sample_data_raw, list):
        sample_data = sample_data_raw

    user_msg = f"Build a transform formula for: {description}"
    if new_field_hint:
        user_msg += f"\nDesired output field name: {new_field_hint}"
    if sample_data:
        user_msg += f"\nUse this sample data exactly: {json.dumps(sample_data[:5], default=str)}"

    # Retrieve relevant RAG context (operation schema + examples) for this description
    system_prompt = _FORMULA_SYSTEM_PROMPT
    try:
        rag_kb = get_rag_kb()
        chunks = await rag_kb.async_search(description, top_k=2)
        rag_context = rag_kb.format_context(chunks)
        if rag_context:
            system_prompt = rag_context + "\n\n" + _FORMULA_SYSTEM_PROMPT
    except Exception as rag_exc:
        logger.debug("RAG search skipped in build_formula: %s", rag_exc)

    try:
        result = await client.chat.completions.create(
            model=model,
            messages=[
                {"role": "system", "content": system_prompt},
                {"role": "user", "content": user_msg},
            ],
            temperature=0.2,
            max_tokens=900,
        )
        _accumulate_usage(usage_acc, result.usage)
        raw = extract_message_content(result.choices[0].message.content)
        formula_config = json.loads(_extract_json(raw))
        return "success", formula_config
    except json.JSONDecodeError as exc:
        return "error", f"LLM returned invalid JSON for formula config: {exc}"
    except Exception as exc:
        logger.error("LLM build formula error: %s", exc, exc_info=True)
        return "error", str(exc)

async def _exec_transform_from_prior(step: dict, confirmed_params: dict, prior_results: dict):
    """Run the transform engine using formula config stored by a prior llm_build_formula step."""
    formula_step_id = step.get("formula_step_id", "build_formula")
    prior = prior_results.get(formula_step_id, {})

    if prior.get("status") != "success":
        return "error", f"Formula step '{formula_step_id}' did not succeed — cannot run transform"

    formula_config = prior.get("value", {})
    if not isinstance(formula_config, dict):
        return "error", f"Formula step result is not a dict: {type(formula_config)}"

    merged = {**confirmed_params}
    for key in ("operation_type", "new_field_name", "input_fields", "operation_params"):
        if formula_config.get(key) is not None:
            merged[key] = formula_config[key]

    # Use sample_data from formula config if none was supplied by the user
    if formula_config.get("sample_data") and not merged.get("sample_data"):
        merged["sample_data"] = formula_config["sample_data"]

    return await _exec_transform(merged, prior_results)


async def _exec_read_config_row(step: dict, confirmed_params: dict, prior_results: dict):
    """Read a specific row from a config parquet sheet by row number or NewFieldName."""
    _ensure_api_root_on_path()
    try:
        import pandas as pd
        from routers.config_search import get_available_types, get_file_path

        lookup_type = (confirmed_params.get("lookup_type") or "new_field_name").strip()
        lookup_value = (confirmed_params.get("lookup_value") or "").strip()
        type_filter = (confirmed_params.get("control_type") or "").strip()
        sheet_name = step.get("sheet", "FieldOperations")

        if not lookup_value:
            return "error", "lookup_value is required"

        types_to_search = [type_filter] if type_filter else get_available_types()
        found_rows = []

        for type_name in types_to_search:
            file_path = get_file_path(type_name, sheet_name)
            if not file_path.exists():
                continue
            try:
                df = pd.read_parquet(file_path)
                if lookup_type == "row_number":
                    try:
                        row_idx = int(lookup_value) - 1  # convert 1-based to 0-based
                        if 0 <= row_idx < len(df):
                            row = df.iloc[row_idx].where(df.iloc[row_idx].notna(), other=None).to_dict()
                            row["_type"] = type_name
                            row["_sheet"] = sheet_name
                            row["_row_number"] = int(lookup_value)
                            found_rows.append(row)
                    except (ValueError, IndexError):
                        pass
                else:
                    # Match by NewFieldName (try common column name variants)
                    field_col = next(
                        (c for c in df.columns if c.lower() in ("newfieldname", "new_field_name", "fieldname", "field_name", "outputfield", "output_field")),
                        None,
                    )
                    if field_col:
                        mask = df[field_col].astype(str).str.lower() == lookup_value.lower()
                        for idx, row_data in df[mask].iterrows():
                            row = row_data.where(row_data.notna(), other=None).to_dict()
                            row["_type"] = type_name
                            row["_sheet"] = sheet_name
                            row["_row_number"] = idx + 1
                            found_rows.append(row)
            except Exception as inner_exc:
                logger.warning("Could not read %s/%s: %s", type_name, sheet_name, inner_exc)
                continue

        if not found_rows:
            return "error", f"No rows found matching {lookup_type}='{lookup_value}' in sheet '{sheet_name}'"

        return "success", {"rows": found_rows, "total": len(found_rows), "sheet": sheet_name, "lookup": {lookup_type: lookup_value}}
    except Exception as exc:
        logger.error("Read config row error: %s", exc, exc_info=True)
        return "error", str(exc)

async def execute_step(step: dict, confirmed_params: dict, prior_results: dict, client, model: str, usage_acc: dict = None):
    ft = step.get("function_type", "")
    ua = usage_acc if usage_acc is not None else {}
    if ft == "file_read":
        return await _exec_read_file(confirmed_params, prior_results)
    elif ft == "file_write":
        return await _exec_write_file(confirmed_params, prior_results)
    elif ft == "file_write_summary":
        return await _exec_write_summary(confirmed_params, prior_results)
    elif ft == "transform_execute":
        return await _exec_transform(confirmed_params, prior_results)
    elif ft == "transform_execute_from_prior":
        return await _exec_transform_from_prior(step, confirmed_params, prior_results)
    elif ft == "config_validate_start":
        return await _exec_config_validate_start(confirmed_params, prior_results)
    elif ft == "poll_until":
        return await _exec_poll_until(confirmed_params, prior_results, step.get("poll_config", {}))
    elif ft == "config_search":
        return await _exec_config_search(confirmed_params, prior_results)
    elif ft == "read_config_row":
        return await _exec_read_config_row(step, confirmed_params, prior_results)
    elif ft == "llm_build_formula":
        return await _exec_llm_build_formula(step, confirmed_params, prior_results, client, model, ua)
    elif ft == "llm_build_formula_from_row":
        return await _exec_llm_build_formula_from_row(step, confirmed_params, prior_results, client, model, ua)
    elif ft == "llm_summarize":
        return "llm_stream", _exec_llm_summarize(step, confirmed_params, prior_results, client, model)
    else:
        return "error", f"Unknown function_type: {ft}"

async def execute_workflow_steps(workflow: dict, confirmed_params: dict, client, model: str, usage_acc: Dict[str, int] = None):
    steps = workflow.get("steps", [])
    prior_results: Dict[str, Any] = {}
    errors = 0
    final_content = ""
    if usage_acc is None:
        usage_acc = {"prompt_tokens": 0, "completion_tokens": 0, "total_tokens": 0}

    for idx, step in enumerate(steps):
        yield _sse({
            "event": "step_start",
            "step_id": step["id"],
            "step_name": step["name"],
            "step_index": idx,
            "total_steps": len(steps),
        })

        status, result = await execute_step(step, confirmed_params, prior_results, client, model, usage_acc)

        if status == "llm_stream":
            gen = result
            accumulated = ""
            async for chunk_type, chunk_content in gen:
                if chunk_type == "chunk":
                    accumulated += chunk_content
                    yield _sse({"event": "llm_chunk", "content": chunk_content})
                elif chunk_type == "done":
                    accumulated = chunk_content
                    final_content = accumulated
                elif chunk_type == "usage":
                    _accumulate_usage(usage_acc, chunk_content)
                elif chunk_type == "error":
                    status = "error"
                    result = chunk_content
                    errors += 1
                    break
            if status == "llm_stream":
                status = "success"
                result = accumulated

        if status == "error":
            errors += 1

        prior_results[step["id"]] = {"status": status, "value": result}

        if isinstance(result, str):
            preview = result[:300]
        elif isinstance(result, (list, dict)):
            preview = json.dumps(result, default=str)[:300]
        else:
            preview = str(result)[:300]

        yield _sse({
            "event": "step_result",
            "step_id": step["id"],
            "step_name": step["name"],
            "status": status,
            "result_preview": preview,
        })

    yield _sse({"event": "final", "content": final_content})
    yield _sse({
        "event": "done",
        "needs_confirm": False,
        "workflow_id": workflow["id"],
        "total_steps_run": len(steps),
        "errors": errors,
        "usage": usage_acc,
    })

async def stream_workflow_chat(request: WorkflowChatRequest):
    client = get_openai_client()
    try:
        data = load_workflows()
        workflows_list = data.get("workflows", [])
        workflows_by_id = {w["id"]: w for w in workflows_list}

        usage_acc: Dict[str, int] = {"prompt_tokens": 0, "completion_tokens": 0, "total_tokens": 0}

        # Phase 4 fast-path: confirmed re-send with all params
        if request.workflow_id and request.confirmed_params is not None:
            workflow = workflows_by_id.get(request.workflow_id)
            if not workflow:
                yield _sse({"event": "error", "error": f"Unknown workflow: {request.workflow_id}"})
                return

            yield _sse({
                "event": "workflow_classified",
                "workflow_id": workflow["id"],
                "workflow_name": workflow["name"],
                "description": workflow["description"],
                "icon": workflow.get("icon", ""),
                "steps": [{"id": s["id"], "name": s["name"]} for s in workflow["steps"]],
                "confidence": 1.0,
            })

            async for chunk in execute_workflow_steps(workflow, request.confirmed_params, client, request.model, usage_acc):
                yield chunk
            return

        # Phase 1 — intent classification
        classification, classify_usage = await classify_intent(client, request.model, request.message, workflows_list)
        _accumulate_usage(usage_acc, classify_usage)
        workflow_id = classification.get("workflow_id")
        confidence = float(classification.get("confidence") or 0.0)

        if not workflow_id or workflow_id not in workflows_by_id or confidence < 0.3:
            # No match — fallback plain LLM response enriched with RAG context
            fallback_system = (
                "You are a helpful assistant for ControlDash, a data orchestration and ETL "
                "control platform. Help the user with questions about controls, transforms, "
                "validations, and configurations."
            )
            try:
                rag_kb = get_rag_kb()
                chunks = await rag_kb.async_search(request.message, top_k=3)
                rag_context = rag_kb.format_context(chunks)
                if rag_context:
                    fallback_system = rag_context + "\n\n" + fallback_system
            except Exception as rag_exc:
                logger.debug("RAG search skipped in fallback: %s", rag_exc)

            result = await client.chat.completions.create(
                model=request.model,
                messages=[
                    {"role": "system", "content": fallback_system},
                    {"role": "user", "content": request.message},
                ],
                temperature=0.7,
                max_tokens=1000,
            )
            _accumulate_usage(usage_acc, result.usage)
            content = extract_message_content(result.choices[0].message.content)
            yield _sse({"event": "content", "content": content})
            yield _sse({"event": "done", "needs_confirm": False, "usage": usage_acc})
            return

        workflow = workflows_by_id[workflow_id]

        yield _sse({
            "event": "workflow_classified",
            "workflow_id": workflow_id,
            "workflow_name": workflow["name"],
            "description": workflow["description"],
            "icon": workflow.get("icon", ""),
            "steps": [{"id": s["id"], "name": s["name"]} for s in workflow["steps"]],
            "confidence": confidence,
        })

        # Phase 2 — param extraction
        params_config = workflow.get("params", [])
        extracted, extract_usage = await extract_params(client, request.model, request.message, params_config)
        _accumulate_usage(usage_acc, extract_usage)

        yield _sse({"event": "params_extracted", "extracted_params": extracted})

        # Phase 3 — check for missing required params
        missing = find_missing_params(params_config, extracted)

        if missing:
            yield _sse({
                "event": "params_required",
                "workflow_id": workflow_id,
                "missing_params": missing,
                "extracted_params": extracted,
                "all_params": params_config,
            })
            yield _sse({"event": "done", "needs_confirm": True, "all_params_present": False, "usage": usage_acc})
        else:
            yield _sse({
                "event": "done",
                "needs_confirm": True,
                "all_params_present": True,
                "workflow_id": workflow_id,
                "extracted_params": extracted,
                "usage": usage_acc,
            })

    except Exception as exc:
        logger.error("Workflow chat error: %s", exc, exc_info=True)
        yield _sse({"event": "error", "error": str(exc)})
