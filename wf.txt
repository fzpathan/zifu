def _accumulate_usage(acc: Dict[str, int], usage) -> None:
    if not usage:
        return
    acc["prompt_tokens"]     += getattr(usage, "prompt_tokens",     0) or 0
    acc["completion_tokens"] += getattr(usage, "completion_tokens", 0) or 0
    acc["total_tokens"]      += getattr(usage, "total_tokens",      0) or 0

class Message(BaseModel):
    role: str
    content: str


class WorkflowChatRequest(BaseModel):
    message: str
    conversation_history: List[Message] = []
    model: str = "nvidia/nemotron-3-nano-omni-30b-a3b-reasoning:free"
    workflow_id: Optional[str] = None
    confirmed_params: Optional[Dict[str, Any]] = None


# ─────────────────────────────────────────────────────────────────────────────
# Phase 1 — Intent classification
# ─────────────────────────────────────────────────────────────────────────────

_CLASSIFICATION_PROMPT = """You are a workflow router. Given the user message and the list of available workflows, identify the single best matching workflow.

Available workflows:
{workflows_summary}

User message: "{message}"

Respond with JSON only (no markdown fences):
{{"workflow_id": "<id_or_null>", "confidence": 0.0}}

Rules:
- workflow_id must exactly match one of the available ids, or be null if nothing matches
- confidence: 0.0-1.0. Use null if confidence < 0.35."""


async def classify_intent(client, model: str, message: str, workflows: list) -> dict:
    parts = []
    for w in workflows:
        ic = w.get("intent_classification", {})
        examples = ic.get("examples", [])[:5]
        keywords = ic.get("keywords", [])[:10]
        parts.append(
            f"- id: {w['id']}\n  name: {w['name']}\n"
            f"  description: {w['description']}\n"
            f"  examples: {examples}\n"
            f"  keywords: {keywords}"
        )
    prompt = _CLASSIFICATION_PROMPT.format(
        workflows_summary="\n".join(parts),
        message=message,
    )
    result = await client.chat.completions.create(
        model=model,
        messages=[{"role": "user", "content": prompt}],
        temperature=0.1,
        max_tokens=120,
    )
    raw = extract_message_content(result.choices[0].message.content)
    try:
        cleaned = re.sub(r"```[a-z]*\n?|```", "", raw).strip()
        return json.loads(cleaned), result.usage
    except Exception:
        return {"workflow_id": None, "confidence": 0.0}, result.usage


# ─────────────────────────────────────────────────────────────────────────────
# Phase 2 — Param extraction
# ─────────────────────────────────────────────────────────────────────────────

_EXTRACTION_PROMPT = """Extract the following parameter values from the user message.

Parameters:
{params_list}

User message: "{message}"

Respond with JSON only (no markdown). Return each parameter key with its extracted string value, or null if not found.
Example: {{"file_path": "/opt/config.xlsx", "control_type": null}}"""


async def extract_params(client, model: str, message: str, params: list) -> dict:
    if not params:
        return {}
    lines = []
    for p in params:
        line = f"- key: {p['key']}, label: {p['label']}, description: {p.get('description', '')}"
        if p.get("options"):
            line += f", allowed_values: {p['options']}"
        if p.get("example"):
            line += f", example: \"{p['example']}\""
        lines.append(line)
    prompt = _EXTRACTION_PROMPT.format(
        params_list="\n".join(lines),
        message=message,
    )
    result = await client.chat.completions.create(
        model=model,
        messages=[{"role": "user", "content": prompt}],
        temperature=0.0,
        max_tokens=500,
    )
    raw = extract_message_content(result.choices[0].message.content)
    try:
        cleaned = re.sub(r"```[a-z]*\n?|```", "", raw).strip()
        return json.loads(cleaned), result.usage
    except Exception:
        return {p["key"]: None for p in params}, result.usage


def find_missing_params(params: list, extracted: dict) -> list:
    missing = []
    for p in params:
        if p.get("required", False):
            val = extracted.get(p["key"])
            if val is None or (isinstance(val, str) and not val.strip()):
                missing.append(p)
    return missing

async def _exec_read_file(confirmed_params: dict, prior_results: dict):
    _ensure_api_root_on_path()
    file_path = (confirmed_params.get("input_file_path") or "").strip()
    if not file_path:
        return "skipped", []

    path = Path(file_path)
    if not path.exists():
        return "error", f"File not found: {file_path}"

    try:
        import pandas as pd
        suffix = path.suffix.lower()
        if suffix in (".xlsx", ".xls"):
            sheet = confirmed_params.get("sheet_name") or 0
            df = pd.read_excel(path, sheet_name=sheet)
        elif suffix == ".csv":
            df = pd.read_csv(path)
        else:
            return "error", f"Unsupported file type '{suffix}'. Use .xlsx, .xls, or .csv"

        df = df.head(500)
        rows = df.where(df.notna(), other=None).to_dict(orient="records")
        return "success", {"rows": rows, "row_count": len(rows), "columns": list(df.columns), "file": file_path}
    except Exception as exc:
        logger.error("Read file error: %s", exc, exc_info=True)
        return "error", str(exc)


async def _exec_write_file(confirmed_params: dict, prior_results: dict):
    _ensure_api_root_on_path()
    output_path = (confirmed_params.get("output_file_path") or "").strip()
    if not output_path:
        return "skipped", "No output path specified"

    transform_result = prior_results.get("execute_transform", {}).get("value")
    if not transform_result or isinstance(transform_result, str):
        return "error", "No transform results available to write"

    try:
        import pandas as pd
        path = Path(output_path)
        path.parent.mkdir(parents=True, exist_ok=True)

        df = pd.DataFrame(transform_result if isinstance(transform_result, list) else [transform_result])
        suffix = path.suffix.lower()
        if suffix == ".csv":
            df.to_csv(path, index=False)
        else:
            if suffix not in (".xlsx", ".xls"):
                path = path.with_suffix(".xlsx")
            df.to_excel(path, index=False)

        return "success", {"written_to": str(path), "rows": len(df), "columns": list(df.columns)}
    except Exception as exc:
        logger.error("Write file error: %s", exc, exc_info=True)
        return "error", str(exc)


async def _exec_write_summary(confirmed_params: dict, prior_results: dict):
    output_path = (confirmed_params.get("summary_output_path") or "").strip()
    if not output_path:
        return "skipped", "No summary output path specified"

    summary = prior_results.get("summarize", {}).get("value", "")
    if not summary:
        return "error", "No summary content to write"

    try:
        path = Path(output_path)
        path.parent.mkdir(parents=True, exist_ok=True)
        suffix = path.suffix.lower()

        if suffix in (".xlsx", ".xls"):
            import pandas as pd
            df = pd.DataFrame({"Validation Summary": [summary]})
            df.to_excel(path, index=False)
        else:
            path.write_text(summary, encoding="utf-8")

        return "success", {"written_to": str(path), "size_chars": len(summary)}
    except Exception as exc:
        logger.error("Write summary error: %s", exc, exc_info=True)
        return "error", str(exc)


async def _exec_transform(confirmed_params: dict, prior_results: dict):
    _ensure_api_root_on_path()
    try:
        import pandas as pd
        from transform_operations.transform_operations import TransformOperations

        ops = TransformOperations()

        # Prefer data read from file over inline JSON
        read_result = prior_results.get("read_input_file", {})
        if read_result.get("status") == "success" and read_result.get("value"):
            file_data = read_result["value"]
            sample_data = file_data.get("rows", []) if isinstance(file_data, dict) else file_data
        else:
            sample_data = confirmed_params.get("sample_data", "[]")
            if isinstance(sample_data, str):
                sample_data = json.loads(sample_data) if sample_data.strip() else []

        if not sample_data:
            return "error", "No sample data provided. Either supply Sample Data JSON or an Input File Path."

        op_params_raw = confirmed_params.get("operation_params") or "{}"
        if isinstance(op_params_raw, str) and op_params_raw.strip():
            try:
                op_params = json.loads(op_params_raw)
            except Exception:
                op_params = {}
        elif isinstance(op_params_raw, dict):
            op_params = op_params_raw
        else:
            op_params = {}

        operation = {
            "Operation": confirmed_params.get("operation_type", ""),
            "NewFieldName": confirmed_params.get("new_field_name", ""),
            "Inputs": confirmed_params.get("input_fields", ""),
            **op_params,
        }

        df = pd.DataFrame(sample_data)
        result_df = ops.execute_operation(df, operation)
        result_df = result_df.where(result_df.notna(), other=None)
        # Convert Timestamps and other non-JSON types to strings
        for col in result_df.columns:
            if result_df[col].dtype.kind in ("M", "m"):  # datetime / timedelta
                result_df[col] = result_df[col].astype(str)
        return "success", result_df.to_dict(orient="records")
    except Exception as exc:
        logger.error("Transform step error: %s", exc, exc_info=True)
        return "error", str(exc)

async def _exec_poll_until(confirmed_params: dict, prior_results: dict, poll_config: dict):
    _ensure_api_root_on_path()
    try:
        from control_execution.control_runner import control_runner

        start_result = prior_results.get("start_validation", {}).get("value", {})
        task_id = start_result.get("task_id") if isinstance(start_result, dict) else None
        if not task_id:
            return "error", "No task_id from start_validation step"

        interval = poll_config.get("interval_seconds", 3)
        max_attempts = poll_config.get("max_attempts", 40)
        done_statuses = poll_config.get("done_statuses", ["completed", "failed", "error"])

        for _ in range(max_attempts):
            status = control_runner.get_task_status(task_id)
            if status.get("status", "") in done_statuses:
                return "success", status
            await asyncio.sleep(interval)

        return "error", f"Polling timed out after {max_attempts * interval}s"
    except Exception as exc:
        logger.error("Poll step error: %s", exc, exc_info=True)
        return "error", str(exc)


async def _exec_config_search(confirmed_params: dict, prior_results: dict):
    _ensure_api_root_on_path()
    try:
        from routers.config_search import (
            _run_parquet_subprocess,
            get_available_types,
            get_file_path,
        )

        query = (confirmed_params.get("query") or "").strip()
        sheet_filter = (confirmed_params.get("sheet") or "").strip()
        type_filter = (confirmed_params.get("control_type") or "").strip()

        if not query:
            return "error", "Search query is required"

        available_sheets = ["InputFiles", "rules", "Enrichment", "FieldOperations", "OutputRule"]
        sheets_to_search = [sheet_filter] if sheet_filter else available_sheets
        types_to_search = [type_filter] if type_filter else get_available_types()

        results_all: List[dict] = []
        for type_name in types_to_search:
            for sheet_name in sheets_to_search:
                file_path = get_file_path(type_name, sheet_name)
                if not file_path.exists():
                    continue
                try:
                    sr = _run_parquet_subprocess("search", {
                        "file_path": str(file_path),
                        "query": query,
                        "column": None,
                        "limit": 100,
                    })
                    rows = sr.get("results", [])
                    for row in rows[:20]:
                        row["_type"] = type_name
                        row["_sheet"] = sheet_name
                    results_all.extend(rows[:20])
                except Exception:
                    continue

        return "success", {"results": results_all, "total": len(results_all), "query": query}
    except Exception as exc:
        logger.error("Config search step error: %s", exc, exc_info=True)
        return "error", str(exc)


async def _exec_llm_summarize(step: dict, confirmed_params: dict, prior_results: dict, client, model: str):
    """Async generator that yields (chunk_type, content) tuples."""
    try:
        prompt_template = step.get("prompt_template", "")

        def _resolve(match):
            key = match.group(1).strip()
            if "." in key:
                step_id, field = key.split(".", 1)
                val = prior_results.get(step_id, {}).get("value", {})
                if isinstance(val, dict):
                    resolved = val.get(field, "")
                elif isinstance(val, list):
                    resolved = json.dumps(val[:15], default=str)
                else:
                    resolved = str(val)
                return str(resolved)[:3000]
            else:
                val = confirmed_params.get(key, "")
                if isinstance(val, (dict, list)):
                    return json.dumps(val, default=str)[:500]
                return str(val) if val is not None else ""

        rendered = re.sub(r"\{\{([^}]+)\}\}", _resolve, prompt_template)

        # Append full prior results for richer context
        extra_parts = []
        for sid, sdata in prior_results.items():
            v = sdata.get("value")
            if v is not None:
                extra_parts.append(f"[{sid}]: {json.dumps(v, default=str)[:3000]}")
        if extra_parts:
            rendered += "\n\nFull step data:\n" + "\n".join(extra_parts)

        stream = await client.chat.completions.create(
            model=model,
            messages=[{"role": "user", "content": rendered}],
            temperature=0.3,
            max_tokens=1500,
            stream=True,
            stream_options={"include_usage": True},
        )
        full = ""
        last_usage = None
        async for chunk in stream:
            if chunk.usage:
                last_usage = chunk.usage
            delta = chunk.choices[0].delta if chunk.choices else None
            if delta and delta.content:
                full += delta.content
                yield "chunk", delta.content
        yield "done", full
        yield "usage", last_usage
    except Exception as exc:
        logger.error("LLM summarize error: %s", exc, exc_info=True)
        yield "error", str(exc)

# Step dispatcher

async def execute_step(step: dict, confirmed_params: dict, prior_results: dict, client, model: str):
    ft = step.get("function_type", "")
    if ft == "file_read":
        return await _exec_read_file(confirmed_params, prior_results)
    elif ft == "file_write":
        return await _exec_write_file(confirmed_params, prior_results)
    elif ft == "file_write_summary":
        return await _exec_write_summary(confirmed_params, prior_results)
    elif ft == "transform_execute":
        return await _exec_transform(confirmed_params, prior_results)
    elif ft == "config_validate_start":
        return await _exec_config_validate_start(confirmed_params, prior_results)
    elif ft == "poll_until":
        return await _exec_poll_until(confirmed_params, prior_results, step.get("poll_config", {}))
    elif ft == "config_search":
        return await _exec_config_search(confirmed_params, prior_results)
    elif ft == "llm_summarize":
        return "llm_stream", _exec_llm_summarize(step, confirmed_params, prior_results, client, model)
    else:
        return "error", f"Unknown function_type: {ft}"

# Step loop
# ─────────────────────────────────────────────────────────────────────────────

async def execute_workflow_steps(workflow: dict, confirmed_params: dict, client, model: str, usage_acc: Dict[str, int] = None):
    steps = workflow.get("steps", [])
    prior_results: Dict[str, Any] = {}
    errors = 0
    final_content = ""
    if usage_acc is None:
        usage_acc = {"prompt_tokens": 0, "completion_tokens": 0, "total_tokens": 0}

    for idx, step in enumerate(steps):
        yield _sse({
            "event": "step_start",
            "step_id": step["id"],
            "step_name": step["name"],
            "step_index": idx,
            "total_steps": len(steps),
        })

        status, result = await execute_step(step, confirmed_params, prior_results, client, model)

        if status == "llm_stream":
            gen = result
            accumulated = ""
            async for chunk_type, chunk_content in gen:
                if chunk_type == "chunk":
                    accumulated += chunk_content
                    yield _sse({"event": "llm_chunk", "content": chunk_content})
                elif chunk_type == "done":
                    accumulated = chunk_content
                    final_content = accumulated
                elif chunk_type == "usage":
                    _accumulate_usage(usage_acc, chunk_content)
                elif chunk_type == "error":
                    status = "error"
                    result = chunk_content
                    errors += 1
                    break
            if status == "llm_stream":
                status = "success"
                result = accumulated

        if status == "error":
            errors += 1

        prior_results[step["id"]] = {"status": status, "value": result}

        if isinstance(result, str):
            preview = result[:300]
        elif isinstance(result, (list, dict)):
            preview = json.dumps(result, default=str)[:300]
        else:
            preview = str(result)[:300]

        yield _sse({
            "event": "step_result",
            "step_id": step["id"],
            "step_name": step["name"],
            "status": status,
            "result_preview": preview,
        })

    yield _sse({"event": "final", "content": final_content})
    yield _sse({
        "event": "done",
        "needs_confirm": False,
        "workflow_id": workflow["id"],
        "total_steps_run": len(steps),
        "errors": errors,
        "usage": usage_acc,
    })

async def stream_workflow_chat(request: WorkflowChatRequest):
    client = get_openai_client()
    try:
        data = load_workflows()
        workflows_list = data.get("workflows", [])
        workflows_by_id = {w["id"]: w for w in workflows_list}

        usage_acc: Dict[str, int] = {"prompt_tokens": 0, "completion_tokens": 0, "total_tokens": 0}

        # Phase 4 fast-path: confirmed re-send with all params
        if request.workflow_id and request.confirmed_params is not None:
            workflow = workflows_by_id.get(request.workflow_id)
            if not workflow:
                yield _sse({"event": "error", "error": f"Unknown workflow: {request.workflow_id}"})
                return

            yield _sse({
                "event": "workflow_classified",
                "workflow_id": workflow["id"],
                "workflow_name": workflow["name"],
                "description": workflow["description"],
                "icon": workflow.get("icon", ""),
                "steps": [{"id": s["id"], "name": s["name"]} for s in workflow["steps"]],
                "confidence": 1.0,
            })

            async for chunk in execute_workflow_steps(workflow, request.confirmed_params, client, request.model, usage_acc):
                yield chunk
            return

        # Phase 1 — intent classification
        classification, classify_usage = await classify_intent(client, request.model, request.message, workflows_list)
        _accumulate_usage(usage_acc, classify_usage)
        workflow_id = classification.get("workflow_id")
        confidence = float(classification.get("confidence") or 0.0)

        if not workflow_id or workflow_id not in workflows_by_id or confidence < 0.3:
            # No match — fallback plain LLM response
            result = await client.chat.completions.create(
                model=request.model,
                messages=[
                    {"role": "system", "content": "You are a helpful assistant for ControlDash, a data orchestration and ETL control platform. Help the user with questions about controls, transforms, validations, and configurations."},
                    {"role": "user", "content": request.message},
                ],
                temperature=0.7,
                max_tokens=1000,
            )
            _accumulate_usage(usage_acc, result.usage)
            content = extract_message_content(result.choices[0].message.content)
            yield _sse({"event": "content", "content": content})
            yield _sse({"event": "done", "needs_confirm": False, "usage": usage_acc})
            return

        workflow = workflows_by_id[workflow_id]

        yield _sse({
            "event": "workflow_classified",
            "workflow_id": workflow_id,
            "workflow_name": workflow["name"],
            "description": workflow["description"],
            "icon": workflow.get("icon", ""),
            "steps": [{"id": s["id"], "name": s["name"]} for s in workflow["steps"]],
            "confidence": confidence,
        })

        # Phase 2 — param extraction
        params_config = workflow.get("params", [])
        extracted, extract_usage = await extract_params(client, request.model, request.message, params_config)
        _accumulate_usage(usage_acc, extract_usage)

        yield _sse({"event": "params_extracted", "extracted_params": extracted})

        # Phase 3 — check for missing required params
        missing = find_missing_params(params_config, extracted)

        if missing:
            yield _sse({
                "event": "params_required",
                "workflow_id": workflow_id,
                "missing_params": missing,
                "extracted_params": extracted,
                "all_params": params_config,
            })
            yield _sse({"event": "done", "needs_confirm": True, "all_params_present": False, "usage": usage_acc})
        else:
            yield _sse({
                "event": "done",
                "needs_confirm": True,
                "all_params_present": True,
                "workflow_id": workflow_id,
                "extracted_params": extracted,
                "usage": usage_acc,
            })

    except Exception as exc:
        logger.error("Workflow chat error: %s", exc, exc_info=True)
        yield _sse({"event": "error", "error": str(exc)})

function overallStatus(stepStatuses, steps) {
    if (!steps || steps.length === 0) return 'pending';
    const statuses = steps.map(s => stepStatuses[s.id] || 'pending');
    if (statuses.some(s => s === 'running')) return 'running';
    if (statuses.some(s => s === 'error')) return 'error';
    const activeStatuses = statuses.filter(s => s !== 'skipped');
    if (activeStatuses.length > 0 && activeStatuses.every(s => s === 'success')) return 'success';
    return 'pending';
}

    const scrollContainerRef = useRef(null);
    const abortControllerRef = useRef(null);

    const [showScrollBtn, setShowScrollBtn] = useState(false);
    const [isConfirmed, setIsConfirmed] = useState(false);
    const [lastUsage, setLastUsage] = useState(null);

    useEffect(() => {
        messagesEndRef.current?.scrollIntoView({ behavior: 'smooth' });
    }, [currentMessages, streamingLlmContent, pendingConfirmation, activeWorkflow]);

    useEffect(() => {
        setIsConfirmed(false);
    }, [activeWorkflow]);

    const handleScroll = useCallback(() => {
        const el = scrollContainerRef.current;
        if (!el) return;
        setShowScrollBtn(el.scrollHeight - el.scrollTop - el.clientHeight > 120);
    }, []);

    const scrollToBottom = () => messagesEndRef.current?.scrollIntoView({ behavior: 'smooth' });

    const streamRequest = useCallback(async (requestBody) => {
        abortControllerRef.current = new AbortController();
        setStreamingLlmContent('');

        // Track accumulated LLM content outside the event loop so it's never
        // affected by React state batching or stale closures.
        let accumulatedContent = '';
        let needsConfirmData = null;

        try {
            const response = await fetch(`${API_BASE_URL}/api/workflow-ai/chat`, {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify(requestBody),
                signal: abortControllerRef.current.signal,
            });

            if (!response.ok) throw new Error(`HTTP error: ${response.status}`);

            const reader = response.body.getReader();
            const decoder = new TextDecoder();
            let buffer = '';

            while (true) {
                const { done, value } = await reader.read();
                if (done) break;

                buffer += decoder.decode(value, { stream: true });
                const lines = buffer.split('\n');
                buffer = lines.pop() || '';

                for (const line of lines) {
                    if (!line.startsWith('data: ')) continue;
                    let data;
                    try { data = JSON.parse(line.slice(6)); } catch { continue; }

                    const evt = data.event;

                    if (evt === 'workflow_classified') {
                        setActiveWorkflow({
                            id: data.workflow_id,
                            name: data.workflow_name,
                            description: data.description,
                            icon: data.icon,
                            steps: data.steps || [],
                        });
                        initStepStatuses(data.steps || []);
                    } else if (evt === 'params_extracted') {
                        setExtractedParams(data.extracted_params || {});
                    } else if (evt === 'params_required') {
                        setExtractedParams(data.extracted_params || {});
                        setPendingConfirmation({
                            workflow_id: data.workflow_id,
                            missing_params: data.missing_params || [],
                            extracted_params: data.extracted_params || {},
                            all_params: data.all_params || {},
                        });
                        setIsLoading(false);
                        return;
                    } else if (evt === 'step_start') {
                        updateStepStatus(data.step_id, 'running');
                    } else if (evt === 'step_result') {
                        const s = data.status;
                        updateStepStatus(data.step_id, s === 'success' ? 'success' : s === 'skipped' ? 'skipped' : 'error');
                    } else if (evt === 'llm_chunk') {
                        // Accumulate locally — keep streaming display alive until fully done
                        accumulatedContent += data.content || '';
                        appendStreamingLlm(data.content || '');
                    } else if (evt === 'final') {
                        // Backend sends the full text here; prefer it over chunk-accumulated version
                        if (data.content) accumulatedContent = data.content;
                        // Do NOT clear streamingLlmContent yet — wait until message is persisted below
                    } else if (evt === 'content') {
                        // Plain LLM fallback (unclassified intent)
                        accumulatedContent += data.content || '';
                        appendStreamingLlm(data.content || '');
                    } else if (evt === 'done') {
                        if (data.usage) setLastUsage(data.usage);
                        if (data.needs_confirm) {
                            needsConfirmData = data;
                        }
                    } else if (evt === 'error') {
                        throw new Error(data.error);
                    }
                }
            }

            // Persist the message AFTER the stream is fully read, then clear the
            // streaming display in the same synchronous block so React batches them
            // into one render — no flicker, no disappearing content.
            if (accumulatedContent) {
                updateCurrentMessages(prev => [...prev, {
                    id: `msg-${Date.now()}-ai`,
                    role: 'assistant',
                    content: accumulatedContent,
                    timestamp: new Date().toISOString(),
                }]);

            setStreamingLlmContent('');

                    workflow_id: needsConfirmData.workflow_id,
                    missing_params: [],
                    extracted_params: params,
                    all_params: params,
                });

            if (error.name !== 'AbortError') {

                    id: `msg-${Date.now()}-error`,
                    role: 'assistant',
                    content: `Error: ${error.message}. Please check that the backend is running.`,
                    timestamp: new Date().toISOString(),
                }]);
            }
        } finally {
            setIsLoading(false);
            setStreamingLlmContent('');
            abortControllerRef.current = null;
        }

const formatDate = (iso) => {
    const d = new Date(iso);
    const now = new Date();
    const diffDays = Math.floor((now - d) / 86400000);
    if (diffDays === 0) return d.toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' });
    if (diffDays === 1) return 'Yesterday';
    if (diffDays < 7) return d.toLocaleDateString([], { weekday: 'short' });
    return d.toLocaleDateString([], { month: 'short', day: 'numeric' });
};

        : 'Describe what you want to do… (e.g. "validate my QA config at /opt/config.xlsx")';

    const adjustHeight = useCallback(() => {
        const el = textareaRef.current;
        if (!el) return;
        el.style.height = 'auto';
        el.style.height = Math.min(el.scrollHeight, 150) + 'px';
    }, []);

    const handleChange = (e) => {
        setMessage(e.target.value);
        adjustHeight();
    };

    const handleSend = useCallback(() => {
        const trimmed = message.trim();
        if (!trimmed || isLoading || pendingConfirmation) return;
        onSendMessage(trimmed);
        setMessage('');
        if (textareaRef.current) {
            textareaRef.current.style.height = 'auto';
        }
    }, [message, isLoading, pendingConfirmation, onSendMessage]);

    const handleKeyDown = (e) => {
        if (e.key === 'Enter' && !e.shiftKey) {
            e.preventDefault();
            handleSend();
        }
    };

    const charCount = message.length;
    const overLimit = charCount > 3000;

    return (
        <div style={{
            padding: '12px 20px 16px',
            borderTop: '1px solid #e2e8f0',
            backgroundColor: 'white',
            flexShrink: 0
        }}>
            <div style={{
                display: 'flex',
                alignItems: 'flex-end',
                gap: '10px',
                border: `1px solid ${overLimit ? '#ef4444' : '#dbe1ea'}`,
                borderRadius: '10px',
                padding: '10px 12px',
                backgroundColor: pendingConfirmation ? '#f8fafc' : 'white',
                boxShadow: '0 1px 3px rgba(15,23,42,0.06)',
                transition: 'border-color 0.15s'
            }}>
                <textarea
                    ref={textareaRef}
                    value={message}
                    onChange={handleChange}
                    onKeyDown={handleKeyDown}
                    placeholder={placeholder}
                    disabled={pendingConfirmation}
                    rows={1}
                    style={{
                        flex: 1,
                        border: 'none',
                        outline: 'none',
                        resize: 'none',
                        fontSize: '14px',
                        color: pendingConfirmation ? '#94a3b8' : '#1f2937',
                        backgroundColor: 'transparent',
                        fontFamily: 'inherit',
                        lineHeight: '1.6',
                        maxHeight: '150px',
                        overflowY: 'auto',
                    }}
                />

                {isLoading ? (
                    <button
                        onClick={onStop}
                        title="Stop generation"
                        style={{
                            display: 'flex', alignItems: 'center', justifyContent: 'center',
                            width: '38px', height: '38px', flexShrink: 0,
                            backgroundColor: '#dc2626', border: 'none', borderRadius: '8px',
                            cursor: 'pointer', color: 'white', transition: 'all 0.15s'
                        }}
                        onMouseEnter={e => { e.currentTarget.style.backgroundColor = '#b91c1c'; }}
                        onMouseLeave={e => { e.currentTarget.style.backgroundColor = '#dc2626'; }}
                    >
                        <Square size={14} fill="white" />
                    </button>
                ) : (
                    <button
                        onClick={handleSend}
                        disabled={!message.trim() || pendingConfirmation || overLimit}
                        title="Send message"
                        style={{
                            display: 'flex', alignItems: 'center', justifyContent: 'center',
                            width: '38px', height: '38px', flexShrink: 0,
                            background: (!message.trim() || pendingConfirmation || overLimit)
                                ? '#e2e8f0'
                                : 'linear-gradient(135deg, #7b0012, #db0011)',
                            border: 'none', borderRadius: '8px',
                            cursor: (!message.trim() || pendingConfirmation || overLimit) ? 'not-allowed' : 'pointer',
                            color: (!message.trim() || pendingConfirmation || overLimit) ? '#94a3b8' : 'white',
                            transition: 'all 0.15s',
                            boxShadow: (!message.trim() || pendingConfirmation || overLimit) ? 'none' : '0 2px 8px rgba(123,0,18,0.25)'
                        }}
                    >
                        <Send size={15} />
                    </button>
                )}
            </div>

            <div style={{ display: 'flex', justifyContent: 'space-between', alignItems: 'center', marginTop: '6px', padding: '0 2px' }}>
                <span style={{ fontSize: '11px', color: '#94a3b8' }}>Shift+Enter for new line</span>
                {charCount > 0 && (
                    <span style={{ fontSize: '11px', color: overLimit ? '#ef4444' : '#94a3b8', fontWeight: overLimit ? '600' : '400' }}>
                        {charCount.toLocaleString()}{overLimit ? ' (over limit)' : ''}
                    </span>
                )}
            </div>
        </div>
    );
}

workflowchatContext

import React, { createContext, useContext, useState, useEffect, useCallback } from 'react';

const STORAGE_KEY = 'workflow-assistant-chats';

const generateId = () => `chat-${Date.now()}-${Math.random().toString(36).substr(2, 9)}`;

const loadFromStorage = () => {
    try {
        const stored = localStorage.getItem(STORAGE_KEY);
        return stored ? JSON.parse(stored) : { chats: [], currentChatId: null };
    } catch {
        return { chats: [], currentChatId: null };
    }
};

const saveToStorage = (data) => {
    try {
        localStorage.setItem(STORAGE_KEY, JSON.stringify(data));
    } catch {}
};

const WorkflowChatContext = createContext(null);

export function WorkflowChatProvider({ children }) {
    const [chats, setChats] = useState([]);
    const [currentChatId, setCurrentChatId] = useState(null);
    const [currentMessages, setCurrentMessages] = useState([]);
    const [isLoading, setIsLoading] = useState(false);
    const [streamingContent, setStreamingContent] = useState('');
    const [model, setModel] = useState('minimax/minimax-m2.5:free');
    const [temperature] = useState(0.7);

    // Workflow-specific state
    const [activeWorkflow, setActiveWorkflow] = useState(null);
    const [extractedParams, setExtractedParams] = useState({});
    const [pendingConfirmation, setPendingConfirmation] = useState(null);
    const [stepStatuses, setStepStatuses] = useState({});
    const [streamingLlmContent, setStreamingLlmContent] = useState('');

    useEffect(() => {
        const data = loadFromStorage();
        setChats(data.chats);
        if (data.currentChatId && data.chats.find(c => c.id === data.currentChatId)) {
            setCurrentChatId(data.currentChatId);
            const chat = data.chats.find(c => c.id === data.currentChatId);
            setCurrentMessages(chat?.messages || []);
        }
    }, []);

    useEffect(() => {
        saveToStorage({ chats, currentChatId });
    }, [chats, currentChatId]);

    const resetWorkflowState = useCallback(() => {
        setActiveWorkflow(null);
        setExtractedParams({});
        setPendingConfirmation(null);
        setStepStatuses({});
        setStreamingLlmContent('');
    }, []);

    const createNewChat = useCallback(() => {
        const newChat = {
            id: generateId(),
            title: 'New Chat',
            createdAt: new Date().toISOString(),
            messages: []
        };
        setChats(prev => [newChat, ...prev]);
        setCurrentChatId(newChat.id);
        setCurrentMessages([]);
        resetWorkflowState();
        return newChat.id;
    }, [resetWorkflowState]);

    const selectChat = useCallback((chatId) => {
        setCurrentChatId(chatId);
        const chat = chats.find(c => c.id === chatId);
        setCurrentMessages(chat?.messages || []);
        resetWorkflowState();
    }, [chats, resetWorkflowState]);

    const deleteChat = useCallback((chatId) => {
        setChats(prev => prev.filter(c => c.id !== chatId));
        if (currentChatId === chatId) {
            setCurrentChatId(null);
            setCurrentMessages([]);
            resetWorkflowState();
        }
    }, [currentChatId, resetWorkflowState]);

    const updateCurrentMessages = useCallback((updater) => {
        setCurrentMessages(prev => {
            const newMessages = typeof updater === 'function' ? updater(prev) : updater;
            setChats(chats => chats.map(chat => {
                if (chat.id === currentChatId) {
                    const firstUserMsg = newMessages.find(m => m.role === 'user');
                    const title = firstUserMsg
                        ? firstUserMsg.content.slice(0, 40) + (firstUserMsg.content.length > 40 ? '...' : '')
                        : chat.title;
                    return { ...chat, messages: newMessages, title };
                }
                return chat;
            }));
            return newMessages;
        });
    }, [currentChatId]);

    const clearCurrentChat = useCallback(() => {
        setCurrentMessages([]);
        setChats(prev => prev.map(chat =>
            chat.id === currentChatId ? { ...chat, messages: [], title: 'New Chat' } : chat
        ));
        resetWorkflowState();
    }, [currentChatId, resetWorkflowState]);

    const exportCurrentChat = useCallback(() => {
        const blob = new Blob([JSON.stringify({ messages: currentMessages, model, exportedAt: new Date().toISOString() }, null, 2)], { type: 'application/json' });
        const url = URL.createObjectURL(blob);
        const link = document.createElement('a');
        link.href = url;
        link.download = `workflow-chat-${new Date().toISOString().split('T')[0]}.json`;
        document.body.appendChild(link);
        link.click();
        document.body.removeChild(link);
        URL.revokeObjectURL(url);
    }, [currentMessages, model]);

    const initStepStatuses = useCallback((steps) => {
        const initial = {};
        steps.forEach(s => { initial[s.id] = 'pending'; });
        setStepStatuses(initial);
    }, []);

    const updateStepStatus = useCallback((stepId, status) => {
        setStepStatuses(prev => ({ ...prev, [stepId]: status }));
    }, []);

    const appendStreamingLlm = useCallback((chunk) => {
        setStreamingLlmContent(prev => prev + chunk);
    }, []);

    const value = {
        chats,
        currentChatId,
        currentMessages,
        isLoading,
        setIsLoading,
        streamingContent,
        setStreamingContent,
        model,
        setModel,
        temperature,
        // Workflow state
        activeWorkflow,
        setActiveWorkflow,
        extractedParams,
        setExtractedParams,
        pendingConfirmation,
        setPendingConfirmation,
        stepStatuses,
        updateStepStatus,
        initStepStatuses,
        streamingLlmContent,
        setStreamingLlmContent,
        appendStreamingLlm,
        // Actions
        createNewChat,
        selectChat,
        deleteChat,
        updateCurrentMessages,
        clearCurrentChat,
        exportCurrentChat,
        resetWorkflowState,
    };

    return (
        <WorkflowChatContext.Provider value={value}>
            {children}
        </WorkflowChatContext.Provider>
    );
}

export function useWorkflowChat() {
    const context = useContext(WorkflowChatContext);
    if (!context) throw new Error('useWorkflowChat must be used within a WorkflowChatProvider');
    return context;
}
